1// openvireo-core for C++: include this header (installed as <ovcore.h>).
2//
3// - ov::version(), the crypto primitives, the functions below the facade (ov v1 crypto, OPAQUE
4// steps, ov::merge_kdbx; README.md), ov::new_client() and the handles (ov::Client,
5// ov::Account, ov::Vault, ov::ServiceAccount, ov::SelfHosted, ov::EventStream, ov::TotpChallenge,
6// ov::LoginStep, ov::CancelToken)
7// come from the generated bridge (ov-ffi-cpp/src/lib.rs.h).
8// - The Host interfaces to implement are in ov-ffi-cpp/include/host.h.
9// - Calls that fail throw rust::Error. Wrap them in ov::call() to get an ov::Error that carries
10// the full OvError (wire code, status, retry-after, details, server code).
11//
12// Every call that can touch the network or run a KDF blocks the calling thread until it is done:
13// call them from a worker thread (QThreadPool / QtConcurrent), never the GUI thread, and cancel
14// from anywhere with CancelToken::cancel() on a clone() of the token. See README.md.

44// Runs `f` (a call into the core) and turns a failure into ov::Error. The core records the
45// failing call's OvError on the calling thread; this reads it right away, on the same thread.
46template <class F>
47auto call(F &&f) -> decltype(std::forward<F>(f)()) {
48 try {
49 return std::forward<F>(f)();
50 } catch (const rust::Error &) {
51 throw Error(take_last_error());
52 }
53}

55// Wipes a byte buffer the core returned (secrets cross as bytes; the caller wipes them).
56inline void wipe(rust::Vec<std::uint8_t> &bytes) {
57 volatile std::uint8_t *p = bytes.data();
58 for (std::size_t i = 0; i < bytes.size(); ++i) {
59 p[i] = 0;
60 }
61 bytes.clear();
62}
