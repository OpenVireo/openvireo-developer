76/// Where a sign-in is.

78pub enum LoginStep {

83}

196    /// Asks the server whether this device's session still stands (a token refresh), so a device
197    /// signed out elsewhere finds out now. Does nothing offline.
198    pub async fn verify_session(&self, cancel: CancelToken) -> Result<(), OvError> {
199        if self.is_offline() || self.inner.tokens.load().await?.is_none() {
200            return Ok(());
201        }
202        match self.refresh_token(&cancel).await {
203            Err(e) if e.is_unreachable() => Ok(()),
204            r => r,
205        }
206    }

208    /// Completes the next time the server ends this session (signed out elsewhere, password changed,
209    /// refresh token reused or expired, account deleted), after the account has locked and dropped
210    /// that user's offline data (SPEC §20). Not for this device's own sign-out. Await it in a loop.
211    /// What is dropped depends on the [`SignedOutReason`]: after an accident
212    /// ([`TokenReused`](SignedOutReason::TokenReused),
213    /// [`SessionExpired`](SignedOutReason::SessionExpired)) the vaults' unsynced and AutoFill edits
214    /// stay for the same user's next sign-in (contract-v1.8).
215    pub async fn signed_out(&self, cancel: CancelToken) -> Result<SignedOutReason, OvError> {
216        let signal = self.inner.signal.clone();
217        let seen = signal.count();
218        let wait = signal.wait(seen);
219        let cancelled = cancel.cancelled();
220        futures::pin_mut!(wait, cancelled);
221        match futures::future::select(wait, cancelled).await {
222            futures::future::Either::Left((reason, _)) => Ok(reason),
223            futures::future::Either::Right(_) => Err(OvError::new(ov_error::OvErrorCode::Aborted, "cancelled")),
224        }
225    }

327    /// Goes back online after working offline: re-reads the account and the vault list. Vaults no
328    /// longer listed are forgotten with their local copies. Fails while still unreachable.
329    pub async fn reconnect(&self, cancel: CancelToken) -> Result<ReconnectResult, OvError> {
330        let session = self.inner.tokens.load().await?.ok_or_else(|| crate::impls::util::signed_out("not signed in"))?;
331        let me = self.call(&cancel, |a, t| a.build_me(t), ov_protocol::auth::parse_me).await?;
332        self.follow_email(&me.email).await?;
333        crate::impls::util::lock(&self.inner.inner).deletion = me.deletion_scheduled_for.clone();
334        self.cache_profile(&session.user_id, &crate::impls::account::CachedProfile { kdf: me.kdf, keyset: me.keyset })
335            .await?;
336        let (vaults, dropped) = self.load_vault_list(&cancel).await?;
337        Ok(ReconnectResult { vaults, dropped })
338    }

497    /// `kdbx` may be under any of the vault's key versions. The merge base is `base` (the server's
498    /// bytes from this device's last sync) or else the one in the cached sync state, which is kept
499    /// in the [BlobCache](crate::host::BlobCache) record `kdbx-file:<vault id>` when the account keeps
500    /// offline data (apart from [`Vault::sync`]'s: the file is its own working copy).
501    /// `server_rolled_back` / `server_lost_database` are errors; sync again with
502    /// [`SyncOptions::accept_server_state`] to continue anyway. A vault shared after the token was
503    /// issued refreshes the token once. Contract-v1.6.
504    pub async fn sync_vault_kdbx(
505        &self,
506        vault_id: String,
507        kdbx: Vec<u8>,
508        base: Option<Vec<u8>>,
509        options: SyncOptions,
510        cancel: CancelToken,
511    ) -> Result<VaultKdbxSyncResult, OvError> {
512        crate::impls::vault_files::sync(self, &vault_id, kdbx, base, options, &cancel).await
513    }

515    /// The session's access token (the JWT for the auth and sync servers; SPEC §9 has one audience),
516    /// refreshed first when it is about to expire, or when `force_refresh` (after a 403 for a vault
517    /// shared since it was issued). For code that still makes its own calls with the core's session;
518    /// it never starts a second session. A refused refresh ends the session (`signed_out`). A secret:
519    /// don't log it. Contract-v1.6.
520    pub async fn access_token(&self, force_refresh: bool, cancel: CancelToken) -> Result<SecretBytes, OvError> {
521        Ok(self.bearer_token(force_refresh, &cancel).await?.as_bytes().to_vec())
522    }

566    /// Live updates (SPEC §11). `vault` events come through as
567    /// [`AccountEvent::VaultChanged`](super::AccountEvent::VaultChanged); on `access` events the
568    /// account refreshes its token (new `vlt` claims), reloads the vault list
569    /// ([`AccountEvent::VaultsChanged`](super::AccountEvent::VaultsChanged)) and reconnects;
570    /// `deletion_scheduled` / `account_deleted` end the session
571    /// ([`AccountEvent::SessionEnded`](super::AccountEvent::SessionEnded)).
572    pub async fn subscribe(&self, cancel: CancelToken) -> Result<EventStream, OvError> {
573        cancel.check()?;
574        Ok(crate::impls::syncing::stream(self.sync_client(), Some(self.clone())))
575    }
