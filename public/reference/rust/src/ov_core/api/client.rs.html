16/// The entry point of the core: one per app process (or per test).

18pub struct OvClient {

22impl std::fmt::Debug for OvClient {

28impl OvClient {
29    /// A client over the platform's [`Host`]. Does no I/O.
30    pub fn new(config: ClientConfig, host: Host) -> Self {
31        OvClient { inner: Arc::new(ClientState { host, config }) }
32    }

34    /// The settings this client was made with.
35    pub fn config(&self) -> ClientConfig {
36        self.inner.config.clone()
37    }

39    /// Fetches the discovery document `GET {issuer}/.well-known/openvireo-configuration` (SPEC §10)
40    /// and checks every URL in it against SPEC §1 "Transport" (`insecure_url`). With
41    /// [`ClientConfig::offline`], an unreachable server falls back to the copy from the last
42    /// successful call (kept in the [BlobCache](crate::host::BlobCache)).
43    pub async fn discover(&self, issuer: String, cancel: CancelToken) -> Result<ServerConfig, OvError> {
44        crate::impls::client::discover(self, &issuer, &cancel).await
45    }

47    /// Opens a local `.kdbx` as [`Vault::open_local`] does, with this client's Host: edits are

51        kdbx: Vec<u8>,
52        credentials: KdbxCredentials,

58    /// A new empty local database as [`Vault::create_local`] makes it, with this client's Host (its

63        credentials: KdbxCredentials,

69    /// Discovers the server (or uses [`ClientConfig::server`]) and loads a stored session if any:
70    /// the account is then signed in but locked. TS `Account.connect`, Swift `Account.connect`.
71    /// [`Account::is_offline`] is true when discovery fell back to the cached document.
72    pub async fn connect(&self, issuer: String, cancel: CancelToken) -> Result<Account, OvError> {
73        crate::impls::client::connect(self, issuer, cancel).await
74    }

76    /// Accounts this device remembers for switching, most recently used first (SPEC §21).
77    pub async fn remembered_accounts(&self) -> Result<Vec<RememberedAccount>, OvError> {
78        crate::impls::remembered::all(&self.inner.host).await
79    }
80
81    /// Adds or refreshes a remembered account (last used: now).
82    pub async fn remember_account(&self, email: String, issuer: String, user_id: String) -> Result<(), OvError> {
83        let host = &self.inner.host;
84        let mut list = crate::impls::remembered::all(host).await?;
85        let previous = list.iter().position(|a| a.issuer == issuer && a.user_id == user_id).map(|i| list.remove(i));
86        list.push(RememberedAccount {
87            email,
88            issuer,
89            user_id,
90            last_used_at: crate::impls::util::rfc3339(host.clock.now_unix_ms()),
91            checked_at: previous.as_ref().and_then(|p| p.checked_at.clone()),
92            deletion_scheduled_for: previous.and_then(|p| p.deletion_scheduled_for),
93        });
94        crate::impls::remembered::save(host, &list).await
95    }
96
97    /// Records (or clears) a remembered account's erase date (SPEC §23).
98    pub async fn set_remembered_deletion(
99        &self,
100        issuer: String,
101        user_id: String,
102        at: Option<Rfc3339>,
103    ) -> Result<(), OvError> {
104        crate::impls::remembered::update(&self.inner.host, &issuer, &user_id, |a| a.deletion_scheduled_for = at).await
105    }
106
107    /// Records that the server confirmed the account's parked session now.
108    pub async fn mark_remembered_checked(&self, issuer: String, user_id: String) -> Result<(), OvError> {
109        let now = crate::impls::util::rfc3339(self.inner.host.clock.now_unix_ms());
110        crate::impls::remembered::update(&self.inner.host, &issuer, &user_id, |a| a.checked_at = Some(now)).await
111    }
112
113    /// Removes an account from the remembered list. Doesn't sign it out or delete its Secret Key
114    /// (that is [`Account::forget_device`]).
115    pub async fn forget_remembered(&self, issuer: String, user_id: String) -> Result<(), OvError> {
116        let host = &self.inner.host;
117        let mut list = crate::impls::remembered::all(host).await?;
118        list.retain(|a| !(a.issuer == issuer && a.user_id == user_id));
119        crate::impls::remembered::save(host, &list).await
120    }

122    /// Whether a user has an active, parked or no session on this device, with no network
123    /// (SPEC §21). If a crash left the session under both keys, the active one wins and the
124    /// parked copy is removed. TS static `Account.sessionState`, Swift `Account.sessionState(of:)`.
125    pub async fn session_state(&self, issuer: String, user_id: String) -> Result<SessionState, OvError> {
126        crate::impls::client::session_state(&self.inner.host, &issuer, &user_id).await
127    }
128
129    /// Whether a user's offline cache holds edits not yet uploaded (reads no secrets). TS static
130    /// `Account.hasPendingEdits`, Swift `Account.hasUnsyncedEdits`.
131    pub async fn has_pending_edits(&self, user_id: String) -> Result<bool, OvError> {
132        let cache = &self.inner.host.cache;
133        for id in cache.list(&user_id).await? {
134            if cache.load(&user_id, &id).await?.is_some_and(|v| v.pending.is_some()) {
135                return Ok(true);
136            }
137        }
138        Ok(false)
139    }

141    /// What the public share page knows before opening a share (`GET {api}/public/shares/{id}`,
142    /// SPEC §22): expiry, opens left, and the passcode KDF if it needs one. A share that ended or
143    /// never existed is `not_found`. Works without an account.
144    pub async fn share_info(&self, api: String, share_id: String, cancel: CancelToken) -> Result<ShareInfo, OvError> {
145        crate::impls::client::share_info(self, &api, &share_id, &cancel).await
146    }

148    /// Opens a share link (SPEC §22): counts one open on the server and decrypts the payload with

152    pub async fn open_share(

154        api: String,
155        link: SecretBytes,
156        passcode: Option<SecretBytes>,
157        info: Option<ShareInfo>,
158        cancel: CancelToken,
159    ) -> Result<OpenedShare, OvError> {
160        crate::impls::client::open_share(self, &api, link, passcode, info, &cancel).await
161    }

164/// Whether a remembered account's stored erase date (SPEC §23) is more than a day past, so the app
165/// removes it and its Secret Key. `now_ms` is Unix milliseconds. TS `deletionExpired`, Swift
166/// `Account.deletionExpired`.
167pub fn deletion_expired(scheduled_for: Option<Rfc3339>, now_ms: i64) -> bool {
168    let Some(at) = scheduled_for.as_deref().and_then(crate::impls::util::parse_rfc3339) else { return false };
169    now_ms > at + 24 * 3_600_000
