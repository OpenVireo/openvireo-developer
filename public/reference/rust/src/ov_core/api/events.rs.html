10/// What the event stream reports, after the account has acted on it.
11#[derive(Clone, Debug, PartialEq, Eq)]
12pub enum AccountEvent {
13    /// A vault changed on the server (`event: vault`). Sync it if `etag` differs from
14    /// [`Vault::etag`](crate::api::Vault::etag).
15    VaultChanged {
16        /// The vault.
17        vault_id: String,
18        /// Its new ETag.
19        etag: Option<String>,
20        /// Its new revision.
21        revision: Option<u64>,
22        /// The vault was deleted.
23        deleted: bool,
24    },
25    /// Access changed (`event: access`): the token was refreshed and this is the new vault list.
26    VaultsChanged {
27        /// The vault list now.
28        vaults: Vec<VaultInfo>,
29    },
30    /// The session ended because the account was scheduled for deletion or deleted (SPEC §23); the
31    /// stream is closed.
32    SessionEnded {
33        /// Why.
34        reason: SignedOutReason,
35        /// The erase date, for `DeletionScheduled`.
36        at: Option<Rfc3339>,
37    },
38    /// A legacy self-hosted database changed (`event: database`, SPEC §13).
39    DatabaseChanged {
40        /// The database.
41        database_id: String,
42        /// Its new ETag.
43        etag: Option<String>,
44        /// Its new revision.
45        revision: Option<u64>,
46        /// It was deleted.
47        deleted: bool,
48    },
49}
50
51/// An open event stream. It reconnects by itself (with a fresh token, after `exp`, and with backoff
52/// while offline) until closed or dropped.
53#[derive(Clone)]
54pub struct EventStream {
55    pub(crate) inner: Arc<EventStreamState>,
56}
57
58impl std::fmt::Debug for EventStream {
59    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
60        f.write_str("EventStream { .. }")
61    }
62}
63
64impl EventStream {
65    /// The next event; `None` once the stream is closed. Connection errors while reconnecting are
66    /// returned as `Err` and the stream keeps going.
67    pub async fn next(&self, cancel: CancelToken) -> Result<Option<AccountEvent>, OvError> {
68        crate::impls::syncing::next_event(self, &cancel).await
69    }
70
71    /// Reconnects now (e.g. after the app returns to the foreground).
72    pub fn reconnect(&self) {
73        self.inner.stream.reconnect()
74    }
75
76    /// Closes the stream; pending and later [`next`](Self::next) calls return `None`.
77    pub fn close(&self) {
78        self.inner.stream.close()
79    }
80}
