1//! Plans and limits, and In-App Purchase (SPEC §19). Limits only stop new things and the server's
2//! check is what counts: clients use these to explain and grey out, never to refuse on their own.
3
4use ov_error::OvError;
5
6use super::{Account, AppleEligibility, AppleTransactionResult, CancelToken, Plan};
7
8impl Account {
9    /// The account's personal plan (from `GET /me`); `None` on a server without billing.
10    pub async fn plan(&self, cancel: CancelToken) -> Result<Option<Plan>, OvError> {
11        let me = self.call(&cancel, |a, t| a.build_me(t), ov_protocol::auth::parse_me).await?;
12        Ok(me.plan.as_ref().map(crate::impls::convert::plan))
13    }
14
15    /// A vault's sync quota in bytes, from the access token's `vlt` claim (`q`); `None` when only the
16    /// server's own limit applies.
17    pub async fn vault_quota(&self, vault_id: String) -> Result<Option<u64>, OvError> {
18        let Some(token) = self.inner.tokens.access_token_now() else { return Ok(None) };
19        let Ok(d) = ov_crypto::jwt::decode_jwt(&token) else { return Ok(None) };
20        Ok(d.claims.vlt.unwrap_or_default().into_iter().find(|v| v.id == vault_id).and_then(|v| v.q))
21    }
22
23    /// Where to change plan after a `plan_limit` error: the error's `upgradeUrl` when it is a
24    /// `/subscription/` page of this account's auth site (same scheme or https, host and port), else
25    /// the site's `subscription/pricing` page. `None` when the issuer has no host.
26    pub fn upgrade_url(&self, error: OvError) -> Option<String> {
27        crate::impls::plan::upgrade_url(error.details.as_deref(), &self.inner.config.issuer)
28    }
29
30    /// Whether the App Store may sell a plan to this user (`org_id` `None`) or to one of their orgs.
31    pub async fn apple_eligibility(
32        &self,
33        org_id: Option<String>,
34        cancel: CancelToken,
35    ) -> Result<AppleEligibility, OvError> {
36        let e = self
37            .call(
38                &cancel,
39                |a, t| a.build_apple_eligibility(t, org_id.as_deref()),
40                ov_protocol::auth::parse_apple_eligibility,
41            )
42            .await?;
43        Ok(AppleEligibility { eligible: e.eligible, reason: e.reason })
44    }
45
46    /// Reports a StoreKit transaction's JWS to the auth server; returns the plan now.
47    pub async fn report_apple_transaction(
48        &self,
49        jws: String,
50        cancel: CancelToken,
51    ) -> Result<AppleTransactionResult, OvError> {
52        let r = self
53            .call(&cancel, |a, t| a.build_report_apple_transaction(t, &jws), ov_protocol::auth::parse_apple_transaction)
54            .await?;
55        Ok(AppleTransactionResult { subject: r.subject, plan: crate::impls::convert::plan(&r.plan) })
56    }
57}
