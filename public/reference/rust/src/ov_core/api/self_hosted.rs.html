1//! [`SelfHosted`]: sync without an account (SPEC §13 `standalone` mode). A device token from a
2//! pairing code, then whole-database sync of password-protected `.kdbx` files against
3//! `databases/<id>` with the same ETag, merge and event logic as account vaults.
4

15/// A pairing code for another device.
16#[derive(Clone, Debug, PartialEq, Eq)]
17pub struct PairingCode {
18    /// `ABCD-EFGH`.
19    pub code: String,
20    /// When it stops working.
21    pub expires_at: Rfc3339,
22}

24/// The result of [`SelfHosted::sync_database`].
25#[derive(Clone, PartialEq, Eq)]
26pub struct SelfHostedSyncResult {
27    /// The merged database, encrypted with the same credentials: save it as the local file.
28    pub kdbx: Vec<u8>,
29    /// What the sync did.
30    pub report: SyncReport,
31}
32
33impl std::fmt::Debug for SelfHostedSyncResult {
34    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
35        f.debug_struct("SelfHostedSyncResult").field("report", &self.report).finish()
36    }
37}

39/// A paired self-hosted sync server.
40#[derive(Clone)]
41pub struct SelfHosted {
42    pub(crate) inner: Arc<SelfHostedState>,
43}
44
45impl std::fmt::Debug for SelfHosted {
46    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
47        f.write_str("SelfHosted { .. }")
48    }
49}

51impl SelfHosted {
52    /// A server with a device token already in hand. Checks the URL (SPEC §1 "Transport").
53    pub fn connect(client: OvClient, server_url: String, device_token: SecretBytes) -> Result<SelfHosted, OvError> {
54        let token = crate::impls::util::secret_text(device_token, "device token")?;
55        SelfHosted::new_impl(client, server_url, token)
56    }
57
58    /// Claims a pairing code (`ABCD-EFGH`) for a device token and keeps the token in the
59    /// [SecureStore](crate::host::SecureStore).
60    pub async fn pair(
61        client: OvClient,
62        server_url: String,
63        code: String,
64        device_name: String,
65        cancel: CancelToken,
66    ) -> Result<SelfHosted, OvError> {
67        SelfHosted::pair_impl(client, server_url, &code, &device_name, &cancel).await
68    }
69
70    /// The server paired earlier, from the stored device token; `None` if not paired.
71    pub async fn restore(client: OvClient, server_url: String) -> Result<Option<SelfHosted>, OvError> {
72        SelfHosted::restore_impl(client, server_url).await
73    }
74
75    /// The server URL.
76    pub fn server_url(&self) -> String {
77        self.inner.server_url.clone()
78    }
79
80    /// `GET /health`, as JSON text. An address that answers but isn't an OpenVireo sync server (a
81    /// web page, a 404, JSON without `status`) is `invalid_request` with
82    /// [`wire_code`](OvError::wire_code) `not_sync_server` (in `server_code`) and the page's status
83    /// (contract-v1.8; before, the page's own status code); an unreachable one stays
84    /// `network_error`, and an error the sync server itself sent (`{error, code}`) is returned as is.
85    pub async fn health(&self, cancel: CancelToken) -> Result<String, OvError> {
86        self.health_impl(&cancel).await
87    }
88
89    /// Downloads `databases/<database_id>` as the server has it (`GET`, no key needed), to save a
90    /// first copy on this device; `None` when the server has no database with that id. The bytes are
91    /// the `.kdbx` as uploaded (encrypted with the database's own credentials). Contract-v1.8.
92    pub async fn download(&self, database_id: String, cancel: CancelToken) -> Result<Option<Vec<u8>>, OvError> {
93        self.download_impl(&database_id, &cancel).await
94    }
95
96    /// The databases on the server.
97    pub async fn list_databases(&self, cancel: CancelToken) -> Result<Vec<BlobMeta>, OvError> {
98        self.list_impl(&cancel).await
99    }
100
101    /// A pairing code for another device.
102    pub async fn create_pairing_code(&self, cancel: CancelToken) -> Result<PairingCode, OvError> {
103        self.pairing_code_impl(&cancel).await
104    }
105
106    /// Syncs a local database with `databases/<database_id>`, keeping the sync state in the
107    /// [BlobCache](crate::host::BlobCache). `kdbx` and the result are encrypted with `credentials`.
108    /// The same as [`sync_database_with`](Self::sync_database_with) with the default
109    /// [`SyncOptions`] (the local copy changed: always downloads).
110    pub async fn sync_database(
111        &self,
112        database_id: String,
113        kdbx: Vec<u8>,
114        credentials: KdbxCredentials,
115        cancel: CancelToken,
116    ) -> Result<SelfHostedSyncResult, OvError> {
117        crate::impls::syncing::sync_database(self, &database_id, &kdbx, &credentials, SyncOptions::default(), &cancel)
118            .await
119    }
120
121    /// [`sync_database`](Self::sync_database) with [`SyncOptions`], as for account vaults:
122    /// - `local_changed: false`: nothing changed in `kdbx` since the last sync, so a HEAD with the
123    ///   last ETag first. When the server didn't change either, nothing is downloaded or decrypted:
124    ///   the result is `kdbx` itself with [`SyncOutcome::Unchanged`](crate::api::SyncOutcome) and no
125    ///   KDF runs. Without a stored sync state it syncs as if changed.
126    /// - `accept_server_state`: "continue anyway" after `server_rolled_back` or
127    ///   `server_lost_database` (forget the stored revision, generation and base).
128    ///
129    /// Contract-v1.7.
130    pub async fn sync_database_with(
131        &self,
132        database_id: String,
133        kdbx: Vec<u8>,
134        credentials: KdbxCredentials,
135        options: SyncOptions,
136        cancel: CancelToken,
137    ) -> Result<SelfHostedSyncResult, OvError> {
138        crate::impls::syncing::sync_database(self, &database_id, &kdbx, &credentials, options, &cancel).await
139    }
140
141    /// The versions the server keeps of `databases/<database_id>`, newest first (`GET
142    /// databases/<id>/history`). Contract-v1.6.
143    pub async fn history(&self, database_id: String, cancel: CancelToken) -> Result<Vec<VaultVersion>, OvError> {
144        self.history_impl(&database_id, &cancel).await
145    }
146
147    /// Downloads one kept version of `databases/<database_id>` and opens it with `credentials` as a
148    /// read-only [`Vault`] ([`Vault::is_read_only`]; compare or restore entries from it, or save it
149    /// with [`Vault::copy_without_remotes`]). Contract-v1.6.
150    pub async fn open_version(
151        &self,
152        database_id: String,
153        revision: u64,
154        credentials: KdbxCredentials,
155        cancel: CancelToken,
156    ) -> Result<Vault, OvError> {
157        self.open_version_impl(&database_id, revision, credentials, &cancel).await
158    }
159
160    /// The legacy `event: database` stream
161    /// ([`AccountEvent::DatabaseChanged`](crate::api::AccountEvent::DatabaseChanged)).
162    pub async fn events(&self, cancel: CancelToken) -> Result<EventStream, OvError> {
163        cancel.check()?;
164        let h = &self.inner.client.host;
165        let client = ov_sync::SyncClient::new(
166            h.http.clone(),
167            h.clock.clone(),
168            self.inner.sync.clone(),
169            std::sync::Arc::new(ov_sync::StaticToken(Some(self.inner.token.to_string()))),
170        );
171        Ok(crate::impls::syncing::stream(client, None))
172    }
