14/// A vault granted to a service account.
15#[derive(Clone, Debug, PartialEq, Eq)]
16pub struct SaVault {
17    /// The vault id.
18    pub id: String,
19    /// Its decrypted name.
20    pub name: String,
21    /// The SA's role.
22    pub role: VaultRole,
23}
24
25/// The result of [`ServiceAccount::inject`].
26#[derive(Clone, PartialEq, Eq)]
27pub struct Injected {
28    /// The template with every `ov://` reference replaced.
29    pub text: SecretBytes,
30    /// The values substituted, for masking output.
31    pub secrets: Vec<SecretBytes>,
32}
33
34impl std::fmt::Debug for Injected {
35    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
36        f.debug_struct("Injected").field("secrets", &self.secrets.len()).finish()
37    }
38}

40/// A signed-in service account. SAs have no refresh token: they exchange the secret again when the
41/// JWT is about to expire.
42#[derive(Clone)]
43pub struct ServiceAccount {
44    pub(crate) inner: Arc<ServiceAccountState>,
45}
46
47impl std::fmt::Debug for ServiceAccount {
48    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
49        f.write_str("ServiceAccount { .. }")
50    }
51}
52
53impl ServiceAccount {
54    /// Exchanges the token's secret for a JWT and unwraps token → KEK → master key → keyset (SPEC
55    /// §9). Discovers the server named in the token unless the client has a
56    /// [`ClientConfig::server`](crate::api::ClientConfig::server).
57    pub async fn from_token(
58        client: OvClient,
59        token: SecretBytes,
60        cancel: CancelToken,
61    ) -> Result<ServiceAccount, OvError> {
62        ServiceAccount::from_token_impl(client, token, &cancel).await
63    }
64
65    /// The server this SA signed in to: the client's
66    /// [`ClientConfig::server`](crate::api::ClientConfig::server), or the one discovered from the
67    /// token's URL (TS `sa.config`). Contract-v1.6.
68    pub fn server_config(&self) -> ServerConfig {
69        self.inner.config.clone()
70    }
71
72    /// The server's issuer (`server_config().issuer`). Contract-v1.6.
73    pub fn issuer(&self) -> String {
74        self.inner.config.issuer.clone()
75    }
76
77    /// The SA's id.
78    pub fn sa_id(&self) -> String {
79        self.inner.sa_id.clone()
80    }
81
82    /// `sa:<id>`.
83    pub fn principal(&self) -> Principal {
84        format!("sa:{}", self.inner.sa_id)
85    }
86
87    /// Vaults granted to this SA, names decrypted (cached; `refresh` asks again).
88    pub async fn vaults(&self, refresh: bool, cancel: CancelToken) -> Result<Vec<SaVault>, OvError> {
89        self.vaults_impl(refresh, &cancel).await
90    }
91
92    /// A vault by id or unique name; `not_found` (with how many matched) otherwise.
93    pub async fn vault(&self, id_or_name: String, cancel: CancelToken) -> Result<SaVault, OvError> {
94        self.vault_impl(&id_or_name, &cancel).await
95    }
96
97    /// Downloads and opens a vault (cached; `refresh` downloads again). Edit it with the [`Vault`]
98    /// methods and upload with [`Vault::sync`] (TS `edit`).
99    pub async fn open_vault(&self, id_or_name: String, refresh: bool, cancel: CancelToken) -> Result<Vault, OvError> {
100        self.open_impl(&id_or_name, refresh, &cancel).await
101    }
102
103    /// Downloads a vault again, merging with the cached copy (TS `pull`).
104    pub async fn pull(&self, id_or_name: String, cancel: CancelToken) -> Result<Vault, OvError> {
105        self.pull_impl(&id_or_name, &cancel).await
106    }
107
108    /// The value of a secret reference `ov://<vault>/[<group>/…]<entry>/<field>` (SPEC §14).
109    pub async fn read(&self, reference: String, cancel: CancelToken) -> Result<SecretBytes, OvError> {
110        Ok(self.read_impl(&reference, &cancel).await?.as_bytes().to_vec())
111    }
112
113    /// Replaces every `ov://…` reference in a template (up to whitespace, quotes or `}`) with its
114    /// value (SPEC §15, `ov inject`).
115    pub async fn inject(&self, template: SecretBytes, cancel: CancelToken) -> Result<Injected, OvError> {
116        let template = crate::impls::util::secret_text(template, "template")?;
117        self.inject_impl(&template, &cancel).await
118    }
119
120    /// Sets the field a secret reference names and uploads (needs `write`). With `create`, a missing
121    /// entry is created.
122    pub async fn write(
123        &self,
124        reference: String,
125        value: SecretBytes,
126        create: bool,
127        cancel: CancelToken,
128    ) -> Result<(), OvError> {
129        let value = crate::impls::util::secret_text(value, "field value")?;
130        self.write_impl(&reference, &value, create, &cancel).await
131    }
132
133    /// The credentials a granted vault's `.kdbx` is under, as
134    /// [`Account::vault_kdbx_credentials`](crate::api::Account::vault_kdbx_credentials): its newest
135    /// vault key as a raw key file. A secret: wipe it. Contract-v1.8.
136    pub async fn vault_kdbx_credentials(
137        &self,
138        vault_id: String,
139        cancel: CancelToken,
140    ) -> Result<KdbxCredentials, OvError> {
141        crate::impls::vault_files::sa_credentials(self, &vault_id, &cancel).await
142    }
143
144    /// Syncs a granted vault held as `.kdbx` bytes, as
145    /// [`Account::sync_vault_kdbx`](crate::api::Account::sync_vault_kdbx) does (needs `write` to
146    /// upload; a read-only grant gets `forbidden` when there is something to upload). An SA keeps no
147    /// offline cache, so the merge base is `base` (the server's bytes from the caller's last sync)
148    /// or none. Contract-v1.8.
149    pub async fn sync_vault_kdbx(
150        &self,
151        vault_id: String,
152        kdbx: Vec<u8>,
153        base: Option<Vec<u8>>,
154        options: SyncOptions,
155        cancel: CancelToken,
156    ) -> Result<VaultKdbxSyncResult, OvError> {
157        crate::impls::vault_files::sa_sync(self, &vault_id, kdbx, base, options, &cancel).await
158    }
159
160    /// Wipes the keyset, vault keys and cached vaults; later calls fail with `signed_out`.
161    pub fn close(&self) {
162        self.close_impl()
163    }
164}
