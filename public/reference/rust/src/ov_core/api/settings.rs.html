1//! Account settings (SPEC §16), two-step sign-in (§8), devices, recovery (§18) and deletion (§23).
2
3use ov_error::OvError;
4
5use super::{
6    Account, CancelToken, DeletionPreview, Device, KdfOverrides, NewAccount, Principal, Profile, RecoveryReset,

10impl Account {
11    // ------------------------------------------------------------------------ profile (§16)
12
13    /// `GET /me`: the signed-in user, plan and orgs, and (contract-v1.8) the KDF parameters and the
14    /// keyset's public keys. Also refreshes the copy kept for offline unlock and follows an email
15    /// change made on another device (Swift `me` and `refreshProfile`). Keys that don't verify are
16    /// `invalid_public_keys`.
17    pub async fn me(&self, cancel: CancelToken) -> Result<Profile, OvError> {
18        self.me_impl(&cancel).await
19    }
20
21    /// Changes the master password (SPEC §16): checks the current one by unwrapping the keyset,
22    /// registers OPAQUE again and wraps the same keyset with the new AUK, so vault keys keep
23    /// working. The Secret Key stays. The server signs out the other devices. Returns the new AUK.
24    pub async fn change_password(
25        &self,
26        current: SecretBytes,
27        next: SecretBytes,
28        kdf: Option<KdfOverrides>,
29        cancel: CancelToken,
30    ) -> Result<SecretBytes, OvError> {
31        self.change_password_impl(current, next, kdf, &cancel).await
32    }
33
34    /// Email change, step 1 (SPEC §16): the server mails an 8-digit code to the new address.
35    pub async fn request_email_change(&self, email: String, cancel: CancelToken) -> Result<(), OvError> {
36        self.request_email_change_impl(&email, &cancel).await
37    }
38
39    /// Email change, step 2: with the mailed code and the master password (checked first), registers
40    /// OPAQUE again under the new email. Keys, Secret Key and vaults stay as they are; what this
41    /// device keeps under the old email moves to the new one.
42    pub async fn change_email(
43        &self,
44        email: String,
45        code: String,
46        password: SecretBytes,
47        cancel: CancelToken,
48    ) -> Result<(), OvError> {
49        self.change_email_impl(&email, &code, password, &cancel).await
50    }
51
52    /// The Secret Key kept on this device, to show the Emergency Kit again (SPEC §2). Needs the
53    /// account unlocked (`signed_out` otherwise).
54    pub async fn secret_key(&self) -> Result<Option<SecretBytes>, OvError> {
55        self.secret_key_impl().await
56    }
57
58    /// A one-time link (valid 60 seconds) that signs the browser in to the auth website and opens
59    /// `next`, a path there such as `/settings` (SPEC §16).
60    pub async fn web_session_url(&self, next: Option<String>, cancel: CancelToken) -> Result<SecretBytes, OvError> {
61        self.web_session_impl(next, &cancel).await
62    }
63
64    // ------------------------------------------------------------------- two-step (§8)
65
66    /// Starts TOTP enrolment: a new secret and `otpauth://` URI to show as a QR code.
67    pub async fn start_two_step(&self, cancel: CancelToken) -> Result<TotpSetup, OvError> {
68        let e = self.call(&cancel, |a, t| a.build_totp_start(t), ov_protocol::auth::parse_totp_start).await?;
69        Ok(TotpSetup { secret: e.secret.into_bytes(), uri: e.otpauth_uri.into_bytes() })
70    }
71
72    /// Turns two-step sign-in on with a code from the authenticator.
73    pub async fn confirm_two_step(&self, code: String, cancel: CancelToken) -> Result<(), OvError> {
74        self.call(&cancel, |a, t| a.build_totp_confirm(t, code.trim()), ov_protocol::parse_empty).await
75    }
76
77    /// Turns two-step sign-in off (needs a current code).
78    pub async fn disable_two_step(&self, code: String, cancel: CancelToken) -> Result<(), OvError> {
79        self.call(&cancel, |a, t| a.build_totp_disable(t, code.trim()), ov_protocol::parse_empty).await
80    }
81
82    // ---------------------------------------------------------------------------- devices
83
84    /// Devices signed in to the account.
85    pub async fn devices(&self, cancel: CancelToken) -> Result<Vec<Device>, OvError> {
86        let l = self.call(&cancel, |a, t| a.build_list_devices(t), ov_protocol::auth::parse_devices).await?;
87        Ok(l.devices.iter().map(crate::impls::convert::device).collect())
88    }
89
90    /// Signs a device out (revokes its session).
91    pub async fn revoke_device(&self, device_id: String, cancel: CancelToken) -> Result<(), OvError> {
92        self.call(&cancel, |a, t| a.build_revoke_device(t, &device_id), ov_protocol::parse_empty).await
93    }
94
95    // --------------------------------------------------------------------- recovery (§18)
96
97    /// What an approved reset will do, for the recovery page. Works signed out.
98    /// `recovery_not_ready` (409, `details.notBefore`) before the link works, `recovery_cancelled`
99    /// (410) after a cancel, `session_expired` (410) once used or expired.
100    pub async fn recover_info(&self, token: SecretBytes, cancel: CancelToken) -> Result<RecoveryReset, OvError> {
101        let token = crate::impls::util::secret_text(token, "reset token")?;
102        self.recover_info_impl(&token, &cancel).await
103    }
104
105    /// Resets the account with an approved `ov_rc_` token (SPEC §18): a new Secret Key, KDF salt and
106    /// keyset as [`register`](Self::register) makes them, a new OPAQUE record, and the old keys and
107    /// vaults archived for 30 days. This device ends up signed in and unlocked; every other device
108    /// is signed out. Show the new Secret Key once. Errors as [`recover_info`](Self::recover_info).
109    pub async fn recover(
110        &self,
111        token: SecretBytes,
112        password: SecretBytes,
113        kdf: Option<KdfOverrides>,
114        trust_device: bool,
115        cancel: CancelToken,
116    ) -> Result<NewAccount, OvError> {
117        self.recover_impl(token, password, kdf, trust_device, &cancel).await
118    }
119
120    /// Whether a recovery archive exists, until when, and whether it is locked (SPEC §18).
121    pub async fn recovery_status(&self, cancel: CancelToken) -> Result<RecoveryStatus, OvError> {
122        let r = self.call(&cancel, |a, t| a.build_recovery_status(t), ov_protocol::auth::parse_recovery_status).await?;
123        Ok(crate::impls::convert::recovery_status(&r))
124    }
125
126    /// Restores the vaults archived by a reset into the current keys (SPEC §18) with the old master
127    /// password and old Secret Key. Needs the account unlocked. Returns how many grants were
128    /// restored. `unauthorized` for a wrong old password; `invalid_secret_key` (before anything is
129    /// restored) when the old Secret Key doesn't open the archived keyset; `restore_locked`,
130    /// `invalid_proof`, `session_expired`, `not_found` from the server.
131    pub async fn restore_from_archive(
132        &self,
133        old_password: SecretBytes,
134        old_secret_key: SecretBytes,
135        cancel: CancelToken,
136    ) -> Result<u32, OvError> {
137        self.restore_from_archive_impl(old_password, old_secret_key, &cancel).await
138    }
139
140    // --------------------------------------------------------------------- deletion (§23)
141
142    /// What deleting this account will do. Vault names are decrypted when the vault list can be read.
143    pub async fn deletion_preview(&self, cancel: CancelToken) -> Result<DeletionPreview, OvError> {
144        self.deletion_preview_impl(&cancel).await
145    }
146
147    /// Schedules this account's deletion (SPEC §23): checks the master password, then the server
148    /// locks the account and erases it on the returned date. `confirm` is the email as the user
149    /// typed it. This device is signed out and its offline cache deleted; the Secret Key stays so the
150    /// user can still restore. A 409 `deletion_blocked` lists the orgs to hand over first, in
151    /// `details` as `{"orgs": [{"id", "name"}, ...]}` (contract-v1.8). While an account is pending
152    /// deletion, any other call the server refuses is 403 `account_pending_deletion` with `details`
153    /// `{"scheduledFor": "<RFC 3339>"}` (contract-v1.8); restore it with
154    /// [`restore_account`](Self::restore_account).
155    pub async fn delete_account(
156        &self,
157        password: SecretBytes,
158        confirm: String,
159        cancel: CancelToken,
160    ) -> Result<Rfc3339, OvError> {
161        self.delete_account_impl(password, &confirm, &cancel).await
162    }
163
164    /// Restores this account while it is pending deletion, after signing in to it (SPEC §23).
165    pub async fn restore_account(&self, password: SecretBytes, cancel: CancelToken) -> Result<(), OvError> {
166        self.restore_account_impl(password, &cancel).await
167    }
168
169    /// Gives a personal vault to someone who already has access (SPEC §23). They become its owner
170    /// with `manage`, this account loses access, and the new owner should rotate the key.
171    pub async fn transfer_vault(&self, vault_id: String, to: Principal, cancel: CancelToken) -> Result<(), OvError> {
172        self.call(&cancel, |a, t| a.build_transfer_vault(t, &vault_id, &to), ov_protocol::auth::parse_vault).await?;
173        self.refresh_token(&cancel).await?;
174        self.load_vault_list(&cancel).await.map(|_| ())
175    }
