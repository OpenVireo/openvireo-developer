62/// `decryption_failed` while unwrapping a keyset means a wrong password or Secret Key
63/// ([`KEYSET_UNWRAP_FAILED`]).
64pub(crate) fn unauthorized_on_decrypt(e: OvError, msg: &str) -> OvError {
65    if e.is(OvErrorCode::DecryptionFailed) {
66        OvError::new(OvErrorCode::Unauthorized, msg).with_status(401).with_details(KEYSET_UNWRAP_FAILED)
67    } else {
68        e
69    }

151    /// `text` opened this account: kept under the key naming this server; an older copy goes.
152    pub(crate) async fn keep_sk(&self, text: &str, email: &str) -> Result<(), OvError> {
153        let store = &self.host().store;
154        store.set(&self.sk_key(email), text.as_bytes()).await?;
155        if self.get_text(&self.older_sk_key(email)).await?.as_deref().map(String::as_str) != Some(text) {
156            return Ok(());
157        }
158        if let Some(device) = self.get_text(&self.older_dev_key(email)).await? {
159            if self.get_text(&self.dev_key(email)).await?.is_none() {
160                store.set(&self.dev_key(email), device.as_bytes()).await?;
161            }
162        }
163        store.delete(&self.older_sk_key(email)).await?;
164        store.delete(&self.older_dev_key(email)).await
165    }

179    /// An access token; when a refresh is refused the session ends here (lock, offline data).
180    pub(crate) async fn bearer_token(&self, force: bool, cancel: &CancelToken) -> Result<Zeroizing<String>, OvError> {
181        match self.inner.tokens.access_token(force, cancel).await {
182            Ok(t) => Ok(t),
183            Err(e) => {
184                if e.is(OvErrorCode::SignedOut) && e.status.is_some() {
185                    self.session_ended(&e).await;
186                }
187                Err(e)
188            }
189        }
190    }

197    /// An authenticated auth-server call, retried once with a fresh token after a 401 (except the
198    /// restore steps, `retries_on_401`).
199    pub(crate) async fn call<T>(
200        &self,
201        cancel: &CancelToken,
202        build: impl Fn(&AuthApi, &str) -> HttpRequest,
203        parse: impl Fn(&HttpResponse) -> Result<T, OvError>,
204    ) -> Result<T, OvError> {
205        let token = self.bearer_token(false, cancel).await?;
206        let req = build(self.auth(), &token);
207        let retry = retries_on_401(&req);
208        let mut res = send(self.host(), req, cancel).await?;
209        if res.status == 401 && retry {
210            let token = self.bearer_token(true, cancel).await?;
211            res = send(self.host(), build(self.auth(), &token), cancel).await?;
212        }
213        parse(&res)
214    }

227    /// The server ended the session (`cause` is the `signed_out` error): lock and drop the user's
228    /// offline data, keeping unsynced and AutoFill edits when it ended by accident (SPEC §20).
229    pub(crate) async fn session_ended(&self, cause: &OvError) {
230        let user = lock(&self.inner.inner).cache_user.take();
231        self.lock_keys();
232        if let Some(u) = user {
233            let _ = self.drop_offline_data(&u, cause).await;
234        }
235    }

237    /// After a session ended with `cause`: everything goes, except each vault's unsynced and
238    /// AutoFill edits when it ended by accident (`token_reused`, `session_expired`; contract-v1.8).
239    pub(crate) async fn drop_offline_data(&self, user_id: &str, cause: &OvError) -> Result<(), OvError> {
240        if ended_by_accident(cause) {
241            self.keep_unsynced_edits(user_id).await
242        } else {
243            self.clear_offline_data(user_id).await
244        }
245    }

247    /// Clears the user's cache but keeps each vault's `pending` and `autofill` parts (still
248    /// encrypted with the vault key), without its sync state and server copy (SPEC §20 "Wiping").
249    async fn keep_unsynced_edits(&self, user_id: &str) -> Result<(), OvError> {
250        if !self.keeps_offline_data() {
251            return Ok(());
252        }
253        let cache = &self.host().cache;
254        let mut kept = Vec::new();
255        for id in cache.list(user_id).await? {
256            if let Some(rec) = cache.load(user_id, &id).await? {
257                if rec.pending.is_some() || rec.autofill.is_some() {
258                    kept.push((
259                        id,
260                        crate::host::CachedVault {
261                            state: Vec::new(),
262                            base: None,
263                            pending: rec.pending,
264                            autofill: rec.autofill,
265                        },
266                    ));
267                }
268            }
269        }
270        cache.clear_user(user_id).await?;
271        for (id, rec) in kept {
272            cache.save(user_id, &id, &rec).await?;
273        }
274        Ok(())
275    }

304    /// Keeps the new session, the Secret Key (when trusted) and the device id.
305    pub(crate) async fn begin(&self, email: &str, s: &w::Session, sk: &[u8; 16], trust: bool) -> Result<(), OvError> {
306        let stored = StoredSession {
307            issuer: self.inner.config.issuer.clone(),
308            user_id: s.user.id.clone(),
309            email: if s.user.email.is_empty() { email.to_owned() } else { s.user.email.clone() },
310            device_id: s.device_id.clone(),
311            refresh_token: Zeroizing::new(s.refresh_token.clone()),
312        };
313        self.inner.tokens.start(stored, &s.access_token, s.expires_in).await?;
314        lock(&self.inner.inner).cache_user = Some(s.user.id.clone());
315        if trust {
316            self.keep_sk(&format_secret_key(sk), email).await?;
317        }
318        self.host().store.set(&self.dev_key(email), s.device_id.as_bytes()).await
319    }

321    /// Derives the AUK and unwraps the keyset into memory; returns the AUK.
322    pub(crate) fn unlock_with(
323        &self,
324        password: &str,
325        kdf: &w::Kdf,
326        wrapped: &w::WrappedKeyset,
327        sk: &[u8; 16],
328    ) -> Result<Zeroizing<[u8; 32]>, OvError> {
329        let params: KdfParams = crypto_kdf(kdf);
330        ov_crypto::auk::check_kdf_params(&params, kdf_min(&self.inner.config))?;
331        let auk = derive_auk(password, &params, sk, self.kdf_opts())?;
332        self.unlock_with_auk_key(&auk, wrapped)?;
333        Ok(auk)
334    }
335
336    /// Unwraps the keyset with an AUK into memory.
337    pub(crate) fn unlock_with_auk_key(&self, auk: &[u8; 32], wrapped: &w::WrappedKeyset) -> Result<(), OvError> {
338        let ks = unwrap_keyset(auk, &crypto_wrapped(wrapped))
339            .map_err(|e| unauthorized_on_decrypt(e, "wrong password or Secret Key"))?;
340        self.lock_keys();
341        lock(&self.inner.inner).keyset = Some(ks);
342        Ok(())
343    }

345    /// Unwraps the keyset from `/me` with this password, which proves it is the master password.
346    pub(crate) async fn check_password(
347        &self,
348        password: &str,
349        sk: &[u8; 16],
350        cancel: &CancelToken,
351    ) -> Result<Keyset, OvError> {
352        let me = self.call(cancel, |a, t| a.build_me(t), ov_protocol::auth::parse_me).await?;
353        let params = crypto_kdf(&me.kdf);
354        ov_crypto::auk::check_kdf_params(&params, kdf_min(&self.inner.config))?;
355        let auk = derive_auk(password, &params, sk, self.kdf_opts())?;
356        unwrap_keyset(&auk, &crypto_wrapped(&me.keyset))
357            .map_err(|e| unauthorized_on_decrypt(e, "the master password is wrong"))
358    }

360    /// The account record for unlocking: `/me`, or offline the cached profile. Returns it and whether
361    /// it came from the cache.
362    pub(crate) async fn unlock_profile(
363        &self,
364        user_id: &str,
365        cancel: &CancelToken,
366    ) -> Result<(CachedProfile, bool), OvError> {
367        match self.call(cancel, |a, t| a.build_me(t), ov_protocol::auth::parse_me).await {
368            Ok(me) => {
369                lock(&self.inner.inner).deletion = me.deletion_scheduled_for.clone();
370                self.follow_email(&me.email).await?;
371                Ok((CachedProfile { kdf: me.kdf, keyset: me.keyset }, false))
372            }
373            Err(e) => {
374                if e.is_unreachable() {
375                    if let Some(p) = self.cached_profile(user_id).await? {
376                        return Ok((p, true));
377                    }
378                }
379                Err(e)
380            }
381        }
382    }

384    /// Moves what this device keeps under the email to the account's current email (SPEC §16).
385    pub(crate) async fn follow_email(&self, email: &str) -> Result<(), OvError> {
386        let Some(session) = self.inner.tokens.load().await? else { return Ok(()) };
387        if email.is_empty() || self.ids(&session.email).id_u == self.ids(email).id_u {
388            return Ok(());
389        }
390        let sk = self.stored_sk(&session.email).await?;
391        let device = self.stored_device(&session.email).await?;
392        let store = &self.host().store;
393        if let Some(sk) = sk {
394            store.set(&self.sk_key(email), sk.as_bytes()).await?;
395        }
396        if let Some(d) = device {
397            store.set(&self.dev_key(email), d.as_bytes()).await?;
398        }
399        for key in [
400            self.sk_key(&session.email),
401            self.dev_key(&session.email),
402            self.older_sk_key(&session.email),
403            self.older_dev_key(&session.email),
404        ] {
405            store.delete(&key).await?;
406        }
407        self.inner.tokens.update_email(email).await
408    }

470    /// `GET /vaults` (offline: the cached list). Returns the list and the vaults no longer listed.
471    pub(crate) async fn load_vault_list(&self, cancel: &CancelToken) -> Result<(Vec<VaultInfo>, Vec<String>), OvError> {
472        self.with_keyset(|_| ())?;
473        let vaults = match self.call(cancel, |a, t| a.build_list_vaults(t), ov_protocol::auth::parse_vaults).await {
474            Ok(l) => l.vaults,
475            Err(e) => {
476                if e.is_unreachable() {
477                    if let Some(cached) = self.cached_wire_vaults().await? {
478                        self.set_offline(true);
479                        let infos = cached.iter().map(|v| self.ingest(v).map(|e| e.info)).collect::<Result<_, _>>()?;
480                        return Ok((infos, vec![]));
481                    }
482                }
483                return Err(e);
484            }
485        };
486        self.set_offline(false);
487        let seen: std::collections::HashSet<&str> = vaults.iter().map(|v| v.id.as_str()).collect();
488        let mut known: Vec<String> = lock(&self.inner.inner).vaults.iter().map(|k| k.id().to_owned()).collect();
489        for v in self.cached_wire_vaults().await?.unwrap_or_default() {
490            if !known.contains(&v.id) {
491                known.push(v.id);
492            }
493        }
494        let dropped: Vec<String> = known.into_iter().filter(|id| !seen.contains(id.as_str())).collect();
495        {
496            let mut g = lock(&self.inner.inner);
497            for id in &dropped {
498                g.vaults.remove(id);
499            }
500        }
501        if self.keeps_offline_data() {
502            if let Some(user) = self.inner.tokens.current().map(|s| s.user_id) {
503                for id in &dropped {
504                    self.host().cache.remove(&user, id).await?;
505                }
506                self.host().cache.save_meta(&user, META_VAULTS, &to_json(&vaults)).await?;
507            }
508        }
509        let infos = vaults.iter().map(|v| self.ingest(v).map(|e| e.info)).collect::<Result<_, _>>()?;
510        Ok((infos, dropped))
511    }

513    /// The keyring entry of a vault: known, `GET /vaults/{id}`, or offline the cached list.
514    pub(crate) async fn entry(&self, vault_id: &str, cancel: &CancelToken) -> Result<VaultEntry, OvError> {
515        if let Some(e) = lock(&self.inner.inner).vaults.get(vault_id) {
516            return Ok(VaultEntry::of(e));
517        }
518        match self.refetch_entry(vault_id, cancel).await {
519            Ok(e) => Ok(e),
520            Err(e) => {
521                if e.is_unreachable() {
522                    if let Some(v) =
523                        self.cached_wire_vaults().await?.unwrap_or_default().into_iter().find(|v| v.id == vault_id)
524                    {
525                        self.set_offline(true);
526                        return self.ingest(&v);
527                    }
528                }
529                Err(e)
530            }
531        }
532    }
533
534    /// `GET /vaults/{id}`: the vault's entry fetched again, replacing the one in the keyring (and
535    /// the cached list). Used when the server copy doesn't open with the keys held: another device
536    /// rotated the vault key since they were fetched.
537    pub(crate) async fn refetch_entry(&self, vault_id: &str, cancel: &CancelToken) -> Result<VaultEntry, OvError> {
538        self.with_keyset(|_| ())?;
539        let v = self.call(cancel, |a, t| a.build_get_vault(t, vault_id), ov_protocol::auth::parse_vault).await?;
540        self.cache_vault(&v).await?;
541        self.ingest(&v)
542    }
543
544    /// The keyring entry from memory or the cached list, without the network.
545    pub(crate) async fn entry_offline(&self, vault_id: &str) -> Result<VaultEntry, OvError> {
546        if let Some(e) = lock(&self.inner.inner).vaults.get(vault_id) {
547            return Ok(VaultEntry::of(e));
548        }
549        let cached = self.cached_wire_vaults().await?.unwrap_or_default();
550        let v = cached.into_iter().find(|v| v.id == vault_id).ok_or_else(|| {
551            OvError::new(OvErrorCode::NotAvailableOffline, format!("vault {vault_id} isn't known on this device"))
552        })?;
553        self.ingest(&v)
554    }
555}

557/// Whether a `signed_out` error says the session ended by accident: the refresh was refused as
558/// `token_reused` or `session_expired` (its `details.reason`), SPEC §20.
559pub(crate) fn ended_by_accident(e: &OvError) -> bool {
560    let reason = e
561        .details
562        .as_deref()
563        .and_then(|d| serde_json::from_str::<serde_json::Value>(d).ok())
564        .and_then(|v| v.get("reason").and_then(|r| r.as_str()).map(str::to_owned));
565    matches!(reason.as_deref(), Some("token_reused" | "session_expired"))
566}
