52pub(crate) fn kdf_min(c: &a::ServerConfig) -> ov_crypto::auk::KdfMin {
53    // The stricter of the server's floor and the SPEC §3 minimum.
54    let base = ov_crypto::auk::KDF_MIN;
55    match c.kdf_min {
56        Some(k) => ov_crypto::auk::KdfMin { m: k.m.max(base.m), t: k.t.max(base.t) },
57        None => base,
58    }
59}

195/// `GET /me` as the facade's profile; `invalid_public_keys` when the keyset's keys don't verify.
196pub(crate) fn profile(m: &w::Me) -> Result<a::Profile, OvError> {
197    let keys = ov_crypto::keyset::trusted_public_keys(&ov_crypto::keyset::PublicKeysJson {
198        enc_pk: m.keyset.enc_pk.clone(),
199        sign_pk: m.keyset.sign_pk.clone(),
200        sig: m.keyset.sig.clone(),
201    })?;
202    Ok(a::Profile {
203        id: m.id.clone(),
204        email: m.email.clone(),
205        name: m.name.clone(),
206        totp_enabled: m.mfa.totp.unwrap_or(false),
207        plan: m.plan.as_ref().map(plan),
208        orgs: m.orgs.iter().flatten().map(org_membership).collect(),
209        deletion_scheduled_for: m.deletion_scheduled_for.clone(),
210        kdf: api_kdf(&m.kdf),
211        public_keys: api_public_keys(&keys),
212    })
213}

354/// `ov_merge` change records → facade records (W1-M follow-up).
355pub(crate) fn merge_change(c: &ov_merge::MergeChange) -> a::MergeChange {
356    a::MergeChange {
357        change: match c.change {
358            ov_merge::ChangeType::Added => a::ChangeType::Added,
359            ov_merge::ChangeType::Modified => a::ChangeType::Modified,
360            ov_merge::ChangeType::Moved => a::ChangeType::Moved,
361            ov_merge::ChangeType::Deleted => a::ChangeType::Deleted,
362            ov_merge::ChangeType::Metadata => a::ChangeType::Metadata,
363        },
364        uuid: c.uuid.as_ref().map(uuid_hex),
365        title: c.title.clone(),
366        group: c.group.clone(),
367        details: c.details.clone(),
368    }
369}

371/// `ov_merge::SyncConflict` → facade record.
372pub(crate) fn sync_conflict(c: &ov_merge::SyncConflict) -> a::SyncConflict {
373    a::SyncConflict {
374        uuid: uuid_hex(&c.uuid),
375        kind: match c.kind {
376            ov_merge::ConflictKind::EditedOnBoth => a::ConflictKind::EditedOnBoth,
377            ov_merge::ConflictKind::DeletedHereEditedThere => a::ConflictKind::DeletedHereEditedThere,
378            ov_merge::ConflictKind::EditedHereDeletedThere => a::ConflictKind::EditedHereDeletedThere,
379        },
380        title: c.title.clone(),
381        fields: c.fields.clone(),
382    }
383}

385/// Parses JSON we stored ourselves; `internal` when it is damaged.
386pub(crate) fn from_stored<T: serde::de::DeserializeOwned>(bytes: &[u8], what: &str) -> Result<T, OvError> {
387    serde_json::from_slice(bytes).map_err(|_| internal(format!("the stored {what} can't be read")))
388}

395    fn every_facade_platform_is_a_known_wire_platform() {
396        use a::Platform::*;
397        for p in [QtLinux, QtMacos, QtWindows, Web, Cli, Connect, Ios, Android] {
398            let wire = wire_platform(p);
399            assert!(!matches!(wire, w::Platform::Other(_)), "{p:?} is not a wire Platform");

401        }
