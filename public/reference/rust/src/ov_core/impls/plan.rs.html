3/// `(scheme, host, port, path)` of an absolute URL; the port is explicit or the scheme's default.
4fn parts(url: &str) -> Option<(String, String, Option<u16>, String)> {
5    let (scheme, rest) = url.trim().split_once("://")?;
6    let scheme = scheme.to_ascii_lowercase();
7    let end = rest.find(['/', '?', '#']).unwrap_or(rest.len());
8    let authority = rest[..end].rsplit_once('@').map_or(&rest[..end], |(_, a)| a).to_ascii_lowercase();
9    let path_end = rest[end..].find(['?', '#']).map_or(rest.len(), |i| end + i);
10    let path = if end < rest.len() { rest[end..path_end].to_owned() } else { "/".to_owned() };
11    let (host, port) = match authority.rsplit_once(':') {
12        Some((h, p)) if !h.is_empty() && !p.contains(']') => (h.to_owned(), Some(p.parse().ok()?)),
13        _ => (authority, None),
14    };
15    if host.is_empty() {
16        return None;
17    }
18    let default = match scheme.as_str() {
19        "https" => Some(443),
20        "http" => Some(80),
21        _ => None,
22    };
23    Some((scheme, host, port.or(default), if path.is_empty() { "/".into() } else { path }))
24}

26/// The error's `upgradeUrl` when it is a `/subscription/` page of the auth site (https or the site's
27/// scheme, same host and port), else `<issuer>/subscription/pricing`; `None` when the issuer has no
28/// host (Swift `Account.upgradeURL`).
29pub(crate) fn upgrade_url(details: Option<&str>, issuer: &str) -> Option<String> {
30    let (site_scheme, site_host, site_port, _) = parts(issuer)?;
31    let candidate = details
32        .and_then(|d| serde_json::from_str::<serde_json::Value>(d).ok())
33        .and_then(|v| v.get("upgradeUrl").or_else(|| v.get("upgrade_url")).and_then(|u| u.as_str()).map(str::to_owned));
34    if let Some(u) = candidate {
35        if let Some((scheme, host, port, path)) = parts(&u) {
36            if (scheme == "https" || scheme == site_scheme)
37                && host == site_host
38                && port == site_port
39                && path.starts_with("/subscription/")
40            {
41                return Some(u);
42            }
43        }
44    }
45    Some(format!("{}/subscription/pricing", issuer.trim().trim_end_matches('/')))
46}

48#[cfg(test)]
49mod tests {
50    use super::*;
51
52    #[test]
53    fn upgrade_urls() {
54        let issuer = "https://accounts.example";
55        let d = r#"{"upgradeUrl":"https://accounts.example/subscription/upgrade?plan=premium"}"#;
56        assert_eq!(upgrade_url(Some(d), issuer).unwrap(), "https://accounts.example/subscription/upgrade?plan=premium");
57        let evil = r#"{"upgradeUrl":"https://evil.example/subscription/x"}"#;
58        assert_eq!(upgrade_url(Some(evil), issuer).unwrap(), "https://accounts.example/subscription/pricing");
59        let other = r#"{"upgradeUrl":"https://accounts.example/settings"}"#;
60        assert_eq!(upgrade_url(Some(other), issuer).unwrap(), "https://accounts.example/subscription/pricing");
61        assert_eq!(upgrade_url(None, "not a url"), None);
62    }
