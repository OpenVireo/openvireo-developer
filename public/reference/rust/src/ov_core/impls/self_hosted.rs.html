1//! [`SelfHosted`] (SPEC §13 `standalone` mode), a port of the SDK's `self-hosted.ts`: a device token
2//! from a pairing code, kept in the SecureStore under `selfhosted:<url>`, and the legacy routes.
3//! Whole-database sync and the `event: database` stream go through `ov-sync`.

17/// `invalid_request` with `server_code` `not_sync_server` and the page's status.
18fn not_sync_server(status: u16) -> OvError {
19    let mut e = OvError::new(ov_error::OvErrorCode::InvalidRequest, "that address isn't an OpenVireo sync server")
20        .with_status(status);
21    e.server_code = Some("not_sync_server".to_owned());
22    e
23}

25/// The SecureStore key of a paired server (TS `SelfHostedSync.tokenKey`).
26pub(crate) fn token_key(url: &str) -> String {
27    format!("selfhosted:{}", url.trim().trim_end_matches('/'))
28}

30impl SelfHosted {
31    pub(crate) fn new_impl(
32        client: OvClient,
33        server_url: String,
34        token: Zeroizing<String>,
35    ) -> Result<SelfHosted, OvError> {
36        let sync = SyncApi::self_hosted(&server_url)?;
37        Ok(SelfHosted { inner: Arc::new(SelfHostedState { client: client.inner, server_url, sync, token }) })
38    }

40    pub(crate) async fn pair_impl(
41        client: OvClient,
42        server_url: String,
43        code: &str,
44        device_name: &str,
45        cancel: &CancelToken,
46    ) -> Result<SelfHosted, OvError> {
47        let sync = SyncApi::self_hosted(&server_url)?;
48        let host = client.inner.host.clone();
49        let claim = parse_pairing_claim(&send(&host, sync.build_claim_pairing_code(code, device_name), cancel).await?)?;
50        let stored = Zeroizing::new(super::util::to_json(&claim));
51        host.store.set(&token_key(&server_url), &stored).await?;
52        SelfHosted::new_impl(client, server_url, Zeroizing::new(claim.token))
53    }
54
55    pub(crate) async fn restore_impl(client: OvClient, server_url: String) -> Result<Option<SelfHosted>, OvError> {
56        let Some(raw) = client.inner.host.store.get(&token_key(&server_url)).await? else { return Ok(None) };
57        let token = serde_json::from_slice::<serde_json::Value>(&raw)
58            .ok()
59            .and_then(|v| v.get("token").and_then(|t| t.as_str()).map(|t| Zeroizing::new(t.to_owned())));
60        match token {
61            Some(t) => SelfHosted::new_impl(client, server_url, t).map(Some),
62            None => Ok(None),
63        }
64    }

66    /// `GET health`; an answer that isn't a sync server's is `not_sync_server` (contract-v1.8).
67    pub(crate) async fn health_impl(&self, cancel: &CancelToken) -> Result<String, OvError> {
68        let res = send(&self.inner.client.host, self.inner.sync.build_health(), cancel).await?;
69        ov_protocol::error::check_redirect(&res)?;
70        let body = serde_json::from_slice::<serde_json::Value>(&res.body).ok();
71        if (200..300).contains(&res.status) {
72            return match body {
73                Some(v) if v.get("status").is_some_and(serde_json::Value::is_string) => Ok(v.to_string()),
74                _ => Err(not_sync_server(res.status)),
75            };
76        }
77        let err = parse_health(&res).expect_err("an error status");
78        // A gateway that can't reach it, or the sync server's own `{error, code}`: as it is
79        if err.is_unreachable() || body.as_ref().and_then(|b| b.get("code")).is_some_and(serde_json::Value::is_string) {
80            return Err(err);
81        }
82        Err(not_sync_server(res.status))
83    }

85    pub(crate) async fn download_impl(
86        &self,
87        database_id: &str,
88        cancel: &CancelToken,
89    ) -> Result<Option<Vec<u8>>, OvError> {
90        cancel.check()?;
91        match self.sync_client().database(database_id).get(None).await? {
92            ov_protocol::sync::GetResult::Ok { data, .. } => Ok(Some(data)),
93            ov_protocol::sync::GetResult::NotFound => Ok(None),
94            ov_protocol::sync::GetResult::NotModified(_) => {
95                Err(OvError::new(ov_error::OvErrorCode::ServerError, "unexpected status 304").with_status(304))
96            }
97        }
98    }

100    pub(crate) async fn list_impl(&self, cancel: &CancelToken) -> Result<Vec<BlobMeta>, OvError> {
101        let req = self.inner.sync.build_list_databases(&self.inner.token);
102        Ok(parse_list_databases(&send(&self.inner.client.host, req, cancel).await?)?.iter().map(blob_meta).collect())
103    }

105    fn sync_client(&self) -> ov_sync::SyncClient {
106        let h = &self.inner.client.host;
107        ov_sync::SyncClient::new(
108            h.http.clone(),
109            h.clock.clone(),
110            self.inner.sync.clone(),
111            Arc::new(ov_sync::StaticToken(Some(self.inner.token.to_string()))),
112        )
113    }

125    pub(crate) async fn open_version_impl(
126        &self,
127        database_id: &str,
128        revision: u64,
129        credentials: KdbxCredentials,
130        cancel: &CancelToken,
131    ) -> Result<Vault, OvError> {
132        let rev = i64::try_from(revision)
133            .map_err(|_| OvError::new(ov_error::OvErrorCode::InvalidRequest, "revision out of range"))?;
134        let key = super::vaults::local_key(&credentials)?;
135        cancel.check()?;
136        let r = self.sync_client().database(database_id).revision(rev).await?;
137        cancel.check()?;
138        let db = ov_kdbx::read(&r.data, &key)?;
139        let host = self.inner.client.host.clone();
140        let v = super::vaults::new_vault(VaultSource::Version { info: None }, Some(host), db, false, None);
141        *lock(&v.inner.etag) = Some(r.version.etag);
142        Ok(v)
143    }

145    pub(crate) async fn pairing_code_impl(&self, cancel: &CancelToken) -> Result<PairingCode, OvError> {
146        let req = self.inner.sync.build_create_pairing_code(&self.inner.token);
