19pub(crate) fn api_payload(p: &ov_share::SharePayload) -> SharePayload {
20    SharePayload {
21        v: p.v,
22        kind: p.kind.clone(),
23        title: p.title.clone(),
24        shared_by: p.shared_by.clone(),
25        fields: p
26            .fields
27            .iter()
28            .map(|f| ShareField {
29                k: f.k.clone(),
30                label: f.label.clone(),
31                value: f.value.as_bytes().to_vec(),
32                protected: f.protected,
33            })
34            .collect(),
35    }
36}

38pub(crate) fn share_payload(p: SharePayload) -> Result<ov_share::SharePayload, OvError> {
39    let fields = p
40        .fields
41        .into_iter()
42        .map(|f| {
43            Ok(ov_share::ShareField {
44                k: f.k,
45                label: f.label,
46                value: secret_text(f.value, "share field")?.to_string(),
47                protected: f.protected,
48            })
49        })
50        .collect::<Result<Vec<_>, OvError>>()?;
51    Ok(ov_share::SharePayload { v: p.v, kind: p.kind, title: p.title, shared_by: p.shared_by, fields })
52}

54pub(crate) fn entry_fields(fields: &[FieldRecord]) -> Vec<ov_share::EntryField> {
55    fields
56        .iter()
57        .map(|f| ov_share::EntryField::new(f.name.clone(), String::from_utf8_lossy(&f.value).into_owned(), f.protected))
58        .collect()
59}

61pub(crate) fn shareable(entry: &EntryRecord) -> Vec<ShareableField> {
62    ov_share::shareable_fields(&entry_fields(&entry.fields))
63        .into_iter()
64        .map(|s| ShareableField {
65            name: s.name.clone(),
66            kind: s.field.k.clone(),
67            label: s.field.label.clone(),
68            value: s.field.value.as_bytes().to_vec(),
69            protected: s.field.protected,
70        })
71        .collect()
72}

75    pub(crate) async fn share_blocked(&self, org_id: Option<String>) -> Result<Option<String>, OvError> {
76        let Some(org_id) = org_id else { return Ok(None) };
77        let me = self.call(&CancelToken::new(), |a, t| a.build_me(t), ov_protocol::auth::parse_me).await?;
78        let Some(org) = me.orgs.unwrap_or_default().into_iter().find(|o| o.id == org_id) else { return Ok(None) };
79        use ov_protocol::types::{MemberStatus, OrgRole, ShareLinkPolicy};
80        Ok(match org.item_share_links.unwrap_or(ShareLinkPolicy::On) {
81            ShareLinkPolicy::Off => Some("Your organization has turned share links off.".into()),
82            ShareLinkPolicy::Managers
83                if !(org.status == MemberStatus::Confirmed && matches!(org.role, OrgRole::Owner | OrgRole::Admin)) =>
84            {
85                Some("In your organization only owners and admins can make share links.".into())
86            }
87            _ => None,
88        })
89    }

91    pub(crate) async fn create_share_impl(
92        &self,
93        payload: SharePayload,
94        options: ShareOptions,
95        cancel: &CancelToken,
96    ) -> Result<CreatedShare, OvError> {
97        let enc_sk = self.with_keyset(|ks| Zeroizing::new(*ks.enc_sk))?;
98        let base = self
99            .share_base_url()
100            .ok_or_else(|| OvError::new(OvErrorCode::Unsupported, "this server has no share page"))?;
101        let payload = share_payload(payload)?;
102        let ShareOptions { expires_in, max_opens, passcode, vault_id, entry_id } = options;
103        let passcode = passcode.map(|p| secret_text(p, "passcode")).transpose()?.filter(|p| !p.is_empty());
104        let (kdf, passcode_key, auth) = match &passcode {
105            Some(pc) => {
106                let kdf = new_kdf_params()?;
107                let key = ov_share::share_passcode_key(pc, &kdf, false)?;
108                let auth = ov_share::share_auth(&key);
109                (Some(kdf), Some(key), Some(b64u(&auth[..])))
110            }
111            None => (None, None, None),
112        };
113        let key = ov_share::new_share_key()?;
114        let req = CreateShareRequest {
115            vault_id: vault_id.clone(),
116            expires_in: i64::from(expires_in),
117            max_opens: max_opens.map(i64::from),
118            kdf: kdf.as_ref().map(wire_kdf),
119            auth,
120        };
121        let share = self.call(cancel, |a, t| a.build_create_share(t, &req), parse_share).await?;
122        let ct = ov_share::encrypt_share_payload(&key, &share.id, &payload, passcode_key.as_deref())?;
123        let reference =
124            ov_share::ShareRef { vault: vault_id, entry: entry_id, title: payload.title.clone().unwrap_or_default() };
125        let r = ov_share::encrypt_share_ref(&enc_sk[..], &share.id, &reference)?;
126        let body = UploadShareRequest { ct: b64u(&ct), reference: r };
127        self.call(cancel, |a, t| a.build_upload_share(t, &share.id, &body), parse_share).await?;
128        let url = ov_share::share_link(&base, &share.id, &key);
129        Ok(CreatedShare { id: share.id, url: url.as_bytes().to_vec(), expires_at: share.expires_at })
130    }

132    pub(crate) async fn list_shares_impl(&self, cancel: &CancelToken) -> Result<Vec<ShareListing>, OvError> {
133        let enc_sk = self.with_keyset(|ks| Zeroizing::new(*ks.enc_sk))?;
134        let list = self.call(cancel, |a, t| a.build_list_shares(t), parse_shares).await?;
135        Ok(list
136            .shares
137            .into_iter()
138            .map(|s| {
139                let reference = s
140                    .reference
141                    .as_deref()
142                    .and_then(|r| ov_share::decrypt_share_ref(&enc_sk[..], &s.id, r).ok())
143                    .map(|r| ShareRef { vault: r.vault.clone(), entry: r.entry.clone(), title: r.title.clone() });
144                ShareListing {
145                    id: s.id,
146                    vault_id: s.vault_id,
147                    reference,
148                    passcode: s.passcode,
149                    max_opens: s.max_opens.map(|n| n.clamp(0, u32::MAX as i64) as u32),
150                    opens: s.opens.clamp(0, u32::MAX as i64) as u32,
151                    opened_at: s.opened_at,
152                    created_at: s.created_at,
153                    expires_at: s.expires_at,
154                    status: s.status.as_str().to_owned(),
155                    ended_at: s.ended_at,
156                    end_reason: s.end_reason.map(|r| r.as_str().to_owned()),
157                }
158            })
159            .collect())
160    }

162    pub(crate) async fn revoke_share_impl(&self, id: &str, cancel: &CancelToken) -> Result<(), OvError> {
163        self.call(cancel, |a, t| a.build_revoke_share(t, id), parse_empty).await
164    }

166    pub(crate) fn share_base_url(&self) -> Option<String> {
167        let c = &self.inner.config;
168        c.share.clone().or_else(|| c.web.as_ref().map(|w| format!("{}/s", w.trim_end_matches('/'))))
169    }
