18fn recipient(pp: &w::PublicPrincipal) -> Result<ShareRecipient, OvError> {
19    let keys = ov_vault::keys::trusted(&ov_vault::keys::public_of_principal(pp))?;
20    Ok(ShareRecipient {
21        principal: pp.principal.clone(),
22        email: pp.email.clone(),
23        name: pp.name.clone(),
24        public_keys: api_public_keys(&keys),
25    })
26}

29    pub(crate) async fn public_principal(
30        &self,
31        who: &Grantee,
32        cancel: &CancelToken,
33    ) -> Result<w::PublicPrincipal, OvError> {
34        match who {
35            Grantee::Principal(p) => {
36                self.call(cancel, |a, t| a.build_public_key(t, p), api::parse_public_principal).await
37            }
38            Grantee::Email(e) => {
39                self.call(cancel, |a, t| a.build_lookup_user(t, e.trim()), api::parse_public_principal).await
40            }
41        }
42    }

45        &self,
46        who: Grantee,
47        display: Option<String>,
48        cancel: &CancelToken,
49    ) -> Result<ShareRecipient, OvError> {
50        let pp = self.public_principal(&who, cancel).await?;

83    async fn share_vault_checked(
84        &self,
85        vault_id: &str,
86        who: Grantee,
87        role: VaultRole,
88        confirmed: Option<crate::api::PublicKeys>,
89        cancel: &CancelToken,
90    ) -> Result<GrantResult, OvError> {
91        let e = self.entry(vault_id, cancel).await?;
92        let pp = self.public_principal(&who, cancel).await?;
93        if let Some(keys) = confirmed {
94            let now = recipient(&pp)?.public_keys;
95            if (now.enc_pk.as_slice(), now.sign_pk.as_slice(), now.sig.as_slice())
96                != (keys.enc_pk.as_slice(), keys.sign_pk.as_slice(), keys.sig.as_slice())
97            {
98                return Err(OvError::new(
99                    OvErrorCode::InvalidPublicKeys,
100                    "the recipient's keys changed since they were shown; check their fingerprint again",
101                ));
102            }
103        }
104        let sealed = ov_vault::grant(&e.vk, &pp, wire_vault_role(role))?;
105        self.call(cancel, |a, t| a.build_grant_access(t, vault_id, std::slice::from_ref(&sealed.grant)), parse_empty)
106            .await?;
107        Ok(GrantResult { principal: pp.principal, fingerprint: sealed.fingerprint })
108    }

110    /// Key rotation (SPEC §7): POST keys (removing `remove` at once) → add the new key → re-key and
111    /// conditional PUT (merge and retry on 412) → activate → re-encrypt the name.
112    pub(crate) async fn rotate_impl(
113        &self,
114        vault_id: &str,
115        remove: Vec<Principal>,
116        cancel: &CancelToken,
117    ) -> Result<SyncReport, OvError> {
118        // The server's current view: a rotation may be pending (SPEC §7, contract-v1.8)
119        let listed = self.call(cancel, |a, t| a.build_get_vault(t, vault_id), api::parse_vault).await?;
120        self.cache_vault(&listed).await?;
121        let mut e = self.ingest(&listed)?;
122        if let Some(pending) = e.info.pending_key_version {
123            if let Some(key) = e.vk.key(pending).map(|k| k.key.clone()) {
124                // This account holds its key (a device stopped before activating): finish it first
125                self.finish_rotation(&e, pending, &key, cancel).await?;
126                e = self.entry(vault_id, cancel).await?;
127            }
128        }
129        let foreign_pending = e.info.pending_key_version.is_some();
130        let me = self.principal_str()?;
131        let access = self.call(cancel, |a, t| a.build_list_access(t, vault_id), api::parse_access).await?.access;
132        let plan = ov_vault::plan_rotation(&e.vk, &me, &access, &remove)?;
133        let mut others = Vec::new();
134        for p in plan.principals_needing_keys() {
135            others.push(self.call(cancel, |a, t| a.build_public_key(t, &p), api::parse_public_principal).await?);
136        }
137        let my_public = self.with_keyset(|ks| ks.public)?;
138        let req = plan.start_request(&my_public, &others)?;
139        match self.call(cancel, |a, t| a.build_start_rotation(t, vault_id, &req), parse_empty).await {
140            Err(err) if foreign_pending && err.is(OvErrorCode::Conflict) => {
141                // Another member's rotation, less than 24 hours old (an older one the server replaces)
142                let mut out = OvError::new(
143                    OvErrorCode::Conflict,
144                    "another member is rotating this vault's key; try again once they're done, or after 24 hours",
145                )
146                .with_status(409);
147                out.server_code = Some("rotation_pending".to_owned());
148                return Err(out);
149            }
150            r => r?,
151        }
152        if let Some(vk) = lock(&self.inner.inner).vaults.get_mut(vault_id) {
153            vk.add_key(plan.new_key());
154        }
155        // One sync from the pending edits or the server's copy, re-keyed before the upload.
156        let e = self.entry(vault_id, cancel).await?;
157        let report = super::syncing::rekey_sync(self, &e, &plan.new_key().key, cancel).await?;
158        let version = i64::from(plan.key_version());
159        self.call(cancel, |a, t| a.build_activate_key(t, vault_id, version), parse_empty).await?;
160        let rename = plan.rename_request(&e.info.name)?;
161        self.call(cancel, |a, t| a.build_rename_vault(t, vault_id, &rename.encrypted_name), api::parse_vault).await?;
162        if let Some(vk) = lock(&self.inner.inner).vaults.get_mut(vault_id) {
163            vk.activate(plan.key_version());
164        }
165        // The cached list must hold the new sealed key, or the re-keyed copy won't open offline.
166        if self.keeps_offline_data() {
167            if let Ok(v) = self.call(cancel, |a, t| a.build_get_vault(t, vault_id), api::parse_vault).await {
168                self.cache_vault(&v).await?;
169                self.ingest(&v)?;
170            }
171        }
172        Ok(report)
173    }

175    /// Finishes a rotation left pending at `version` whose key this account holds: one sync re-keyed
176    /// to it, activate, and the name under the new key (SPEC §7 steps 2 and 3).
177    async fn finish_rotation(
178        &self,
179        e: &super::keyring::VaultEntry,
180        version: u32,
181        key: &[u8],
182        cancel: &CancelToken,
183    ) -> Result<(), OvError> {
184        super::syncing::rekey_sync(self, e, key, cancel).await?;
185        let v = i64::from(version);
186        self.call(cancel, |a, t| a.build_activate_key(t, &e.info.id, v), parse_empty).await?;
187        if let Some(vk) = lock(&self.inner.inner).vaults.get_mut(&e.info.id) {
188            vk.activate(version);
189        }
190        let mut k = zeroize::Zeroizing::new([0u8; 32]);
191        if key.len() == 32 {
192            k.copy_from_slice(key);
193        }
194        let enc = ov_crypto::vault_name::encrypt_vault_name(&k, &e.info.id, &e.info.name)?;
195        let listed = self.call(cancel, |a, t| a.build_rename_vault(t, &e.info.id, &enc), api::parse_vault).await?;
196        self.cache_vault(&listed).await?;
197        self.ingest(&listed)?;
198        Ok(())
199    }

201    /// Rotates every flagged org vault this account manages; failures are reported.
202    pub(crate) async fn rotate_flagged_impl(
203        &self,
204        org_id: &str,
205        cancel: &CancelToken,
206    ) -> Result<RotationReport, OvError> {
207        let mut report = RotationReport { rotated: vec![], failed: vec![] };
208        self.load_vault_list(cancel).await?;
209        let flagged: Vec<String> =
210            lock(&self.inner.inner).vaults.flagged_for_rotation(org_id).iter().map(|v| v.id().to_owned()).collect();
211        for id in flagged {
212            match self.rotate_impl(&id, vec![], cancel).await {
213                Ok(_) => report.rotated.push(id),
214                Err(error) => report.failed.push(RotationFailure { vault_id: id, error }),
215            }
216        }
217        Ok(report)
218    }

220    /// The member as the server lists it (with the public keys the facade record doesn't keep raw).
221    async fn wire_member(&self, org_id: &str, user_id: &str, cancel: &CancelToken) -> Result<w::Member, OvError> {
222        let members = self.call(cancel, |a, t| a.build_list_members(t, org_id), api::parse_members).await?.members;
223        members
224            .into_iter()
225            .find(|m| m.user_id == user_id)
226            .ok_or_else(|| OvError::new(OvErrorCode::NotFound, "no such member").with_status(404))
227    }

229    /// The org's SA master keys this account holds, sealed to another owner/admin.
230    async fn sealed_sa_master_keys(
231        &self,
232        org_id: &str,
233        principal: &str,
234        enc_pk: &[u8; 32],
235        cancel: &CancelToken,
236    ) -> Result<Vec<w::SaMasterGrant>, OvError> {
237        let listed =
238            self.call(cancel, |a, t| a.build_list_service_accounts(t, org_id), api::parse_service_accounts).await?;
239        let mut sas = Vec::new();
240        for sa in listed.service_accounts {
241            if sa.master_key.is_some() {
242                sas.push(sa);
243            } else {
244                sas.push(
245                    self.call(cancel, |a, t| a.build_get_service_account(t, &sa.id), api::parse_service_account)
246                        .await?,
247                );
248            }
249        }
250        let g = lock(&self.inner.inner);
251        let ks = g.keyset.as_ref().ok_or_else(|| signed_out("the account is locked"))?;
252        ov_vault::seal_sa_master_keys(ks, &sas, principal, enc_pk)
253    }

255    /// Confirms an accepted member (SPEC §12): owners and admins get every org vault this account
256    /// manages (`manage`) and the org's SA master keys; a member gets `vaults`.
257    pub(crate) async fn confirm_member_impl(
258        &self,
259        org_id: &str,
260        member: Member,
261        vaults: Vec<VaultGrantRequest>,
262        cancel: &CancelToken,
263    ) -> Result<Member, OvError> {
264        let wm = self.wire_member(org_id, &member.user_id, cancel).await?;
265        if wm.public.is_none() {
266            return Err(
267                OvError::new(OvErrorCode::InvalidRequest, "this member has no public keys yet").with_status(400)
268            );
269        }
270        let target = ov_vault::member_keys(&wm)?;
271        let picked: Vec<ov_vault::VaultChoice> = vaults
272            .iter()
273            .map(|v| ov_vault::VaultChoice { vault_id: v.vault_id.clone(), role: wire_vault_role(v.role) })
274            .collect();
275        self.load_vault_list(cancel).await?;
276        let missing = ov_vault::missing_vaults(&lock(&self.inner.inner).vaults, &picked);
277        for id in missing {
278            self.entry(&id, cancel).await?;
279        }
280        let sa_keys = if target.manager {
281            Some(self.sealed_sa_master_keys(org_id, &target.principal, &target.public.enc_pk, cancel).await?)
282        } else {
283            None
284        };
285        let body = {
286            let g = lock(&self.inner.inner);
287            ov_vault::confirm_member_request(org_id, &target, &g.vaults, &picked, sa_keys)?
288        };
289        let m = self
290            .call(cancel, |a, t| a.build_confirm_member(t, org_id, &member.user_id, &body), api::parse_member)
291            .await?;
292        Ok(super::convert::member(&m))
293    }

295    /// Changes a member's role; a promotion hands over the SA master keys, then `manage` on every org
296    /// vault this account manages.
297    pub(crate) async fn set_member_role_impl(
298        &self,
299        org_id: &str,
300        member: Member,
301        role: OrgRole,
302        cancel: &CancelToken,
303    ) -> Result<Member, OvError> {
304        let wm = self.wire_member(org_id, &member.user_id, cancel).await?;
305        let wrole = wire_org_role(role);
306        let promoted = ov_vault::promotion(&wm, &wrole)?;
307        let sa_keys = match &promoted {
308            Some(k) => Some(self.sealed_sa_master_keys(org_id, &k.principal, &k.public.enc_pk, cancel).await?),
309            None => None,
310        };
311        let body = ov_vault::set_member_role_request(wrole, sa_keys);
312        let m = self
313            .call(cancel, |a, t| a.build_set_member_role(t, org_id, &member.user_id, &body), api::parse_member)
314            .await?;
315        if let Some(k) = promoted {
316            self.load_vault_list(cancel).await?;
317            let managed: Vec<String> =
318                lock(&self.inner.inner).vaults.managed_org_vaults(org_id).iter().map(|v| v.id().to_owned()).collect();
319            for id in managed {
320                self.share_vault_impl(&id, Grantee::Principal(k.principal.clone()), VaultRole::Manage, cancel).await?;
321            }
322        }
323        Ok(super::convert::member(&m))
324    }

326    /// Creates a service account (SPEC §9): keyset and master key made here, the master key sealed
327    /// to every owner and admin, the vaults granted, the first token returned.
328    pub(crate) async fn create_service_account_impl(
329        &self,
330        org_id: &str,
331        input: NewServiceAccount,
332        cancel: &CancelToken,
333    ) -> Result<CreatedServiceAccount, OvError> {
334        let me = self.principal_str()?;
335        let members = self.call(cancel, |a, t| a.build_list_members(t, org_id), api::parse_members).await?.members;
336        let spec = ov_vault::ServiceAccountSpec {
337            name: input.name.clone(),
338            expires_at: input.expires_at.clone(),
339            secret_name: input.secret_name.clone(),
340        };
341        let nsa = {
342            let g = lock(&self.inner.inner);
343            let ks = g.keyset.as_ref().ok_or_else(|| signed_out("the account is locked"))?;
344            ov_vault::new_service_account(&self.inner.config.issuer, &me, ks, &members, &spec)?
345        };
346        let sa = self
347            .call(cancel, |a, t| a.build_create_service_account(t, org_id, &nsa.request), api::parse_service_account)
348            .await?;
349        for g in &input.vaults {
350            let e = self.entry(&g.vault_id, cancel).await?;
351            let grant = nsa.vault_grant(&e.vk, wire_vault_role(g.role))?;
352            self.call(cancel, |a, t| a.build_grant_access(t, &g.vault_id, std::slice::from_ref(&grant)), parse_empty)
353                .await?;
354        }
355        Ok(CreatedServiceAccount {
356            service_account: super::convert::service_account(&sa),
357            token: nsa.token.as_bytes().to_vec(),
358        })
359    }

361    /// Another secret for an SA (owner/admin): the master key unsealed here and wrapped for it.
362    pub(crate) async fn issue_secret_impl(
363        &self,
364        sa_id: &str,
365        name: Option<String>,
366        expires_at: Option<Rfc3339>,
367        cancel: &CancelToken,
368    ) -> Result<IssuedSecret, OvError> {
369        let sa = self.call(cancel, |a, t| a.build_get_service_account(t, sa_id), api::parse_service_account).await?;
370        let issued = {
371            let g = lock(&self.inner.inner);
372            let ks = g.keyset.as_ref().ok_or_else(|| signed_out("the account is locked"))?;
373            ov_vault::issue_sa_secret(&self.inner.config.issuer, ks, &sa, name.as_deref(), expires_at.as_deref())?
374        };
375        self.call(cancel, |a, t| a.build_issue_sa_secret(t, sa_id, &issued.body), parse_empty).await?;
376        Ok(IssuedSecret {
377            token: issued.token.as_bytes().to_vec(),
378            id: issued.body.id.clone(),
379            prefix: issued.body.prefix.clone(),
380        })
381    }
