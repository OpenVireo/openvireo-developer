1//! Sync on the facade, over `ov-sync`: the token sources, account and service-account vault sync
2//! through `VaultSession`, self-hosted database sync, and the event streams.
3//!
4//! The Host has no timer, so the core never sleeps. When `ov_sync::EventStream` says
5//! `Waiting { retry_in_ms }`, [`EventStream::next`] returns `Err(network_error)` with
6//! `retry_after` (seconds, rounded up) and `details: {"retryInMs": n}`; the platform waits that long
7//! and calls `next` again (the contract's "connection errors while reconnecting are returned as
8//! `Err` and the stream keeps going").

92pub(crate) fn report(r: &ov_sync::SyncResult) -> SyncReport {
93    let conflicts: Vec<_> = r.conflicts.iter().map(sync_conflict).collect();
94    SyncReport {
95        outcome: match r.outcome {
96            ov_sync::SyncOutcome::Unchanged => SyncOutcome::Unchanged,
97            ov_sync::SyncOutcome::Pulled => SyncOutcome::Pulled,
98            ov_sync::SyncOutcome::Uploaded => SyncOutcome::Uploaded,
99            ov_sync::SyncOutcome::Created => SyncOutcome::Created,
100        },
101        conflicts,
102        changes: r.changes.iter().map(merge_change).collect(),
103        attempts: r.attempts,
104        etag: Some(r.state.etag.clone()).filter(|e| !e.is_empty()),
105        clock_offset_ms: r.clock_offset_ms,
106        clock_skewed: r.clock_skewed(),
107    }
108}

110/// The keys of the vault behind an account or SA vault handle, newest first.
111async fn vault_keys(v: &Vault) -> Result<(String, Vec<KeyVersion>), OvError> {
112    match &v.inner.source {
113        VaultSource::Account { account, info } => {
114            let id = lock(info).id.clone();
115            let e = account.entry_offline(&id).await?;
116            Ok((id, e.keys().to_vec()))
117        }
118        VaultSource::ServiceAccount { sa, info } => {
119            let sa = sa.upgrade().ok_or_else(|| super::util::signed_out("service account closed"))?;
120            let sa = ServiceAccount { inner: sa };
121            let e = sa.entry(&info.id, &CancelToken::new()).await?;
122            Ok((info.id.clone(), e.keys().to_vec()))
123        }
124        _ => Err(OvError::new(OvErrorCode::InvalidRequest, "this vault has no server")),
125    }
126}

128/// The keys of the vault behind a handle fetched again from the server, after its copy didn't open
129/// with the keys held: another device rotated the vault key since they were fetched. The account's
130/// keyring (or the service account's list) and the handle's record follow.
131async fn refetch_keys(v: &Vault, cancel: &CancelToken) -> Result<Vec<KeyVersion>, OvError> {
132    match &v.inner.source {
133        VaultSource::Account { account, info } => {
134            let id = lock(info).id.clone();
135            let e = account.refetch_entry(&id, cancel).await?;
136            *lock(info) = e.info.clone();
137            Ok(e.keys().to_vec())
138        }
139        VaultSource::ServiceAccount { sa, info } => {
140            let sa = sa.upgrade().ok_or_else(|| super::util::signed_out("service account closed"))?;
141            let sa = ServiceAccount { inner: sa };
142            sa.entries(true, cancel).await?;
143            Ok(sa.entry(&info.id, cancel).await?.keys().to_vec())
144        }
145        _ => Err(OvError::new(OvErrorCode::InvalidRequest, "this vault has no server")),
146    }
147}

149/// Syncs an account or SA vault (conditional PUT, merge and retry on 412), re-keying to `rekey_to`
150/// before the upload (rotation). The working copy, etag, conflicts and the account's clock offset
151/// follow the result. Offline, the working copy goes to the cache's pending edits.
152pub(crate) async fn sync_vault(
153    v: &Vault,
154    options: SyncOptions,
155    rekey_to: Option<&[u8; 32]>,
156    cancel: &CancelToken,
157) -> Result<SyncReport, OvError> {
158    sync_vault_with(v, options, rekey_to, false, cancel).await
159}

161/// [`sync_vault`], where `remote_is_base`: the vault was just downloaded, so the first attempt
162/// uploads against that copy instead of downloading it again (a 412 downloads).
163///
164/// Edits made to the handle while a round runs are merged into its result three ways (the copy the
165/// round started from is the base) and uploaded by up to two more rounds, each against the copy the
166/// round before left on the server (contract-v1.8). A later round that fails leaves them as
167/// unsynced edits and the call returns what the earlier rounds did.
168pub(crate) async fn sync_vault_with(
169    v: &Vault,
170    options: SyncOptions,
171    rekey_to: Option<&[u8; 32]>,
172    remote_is_base: bool,
173    cancel: &CancelToken,
174) -> Result<SyncReport, OvError> {
175    const ROUNDS: usize = 3;
176    cancel.check()?;
177    let (id, mut keys) = vault_keys(v).await?;
178    let client = match &v.inner.source {
179        VaultSource::Account { account, .. } => account.sync_client(),
180        VaultSource::ServiceAccount { sa, .. } => {
181            let sa = sa.upgrade().ok_or_else(|| super::util::signed_out("service account closed"))?;
182            ServiceAccount { inner: sa }.sync_client()
183        }
184        _ => return Err(OvError::new(OvErrorCode::InvalidRequest, "this vault has no server")),
185    };
186    let blob = client.vault(&id);
187    let mut session = v.inner.session.lock().await;
188    if session.key.is_none() {
189        let newest = keys
190            .iter()
191            .max_by_key(|k| k.version)
192            .ok_or_else(|| OvError::new(OvErrorCode::NoVaultKey, "no vault key available"))?;
193        session.key = Some(vault_credentials(&newest.key)?);
194    }
195    let cw = CancelWrap(cancel.clone());
196    let rekey = rekey_to.map(|k| Zeroizing::new(k.to_vec()));
197    let mut total: Option<SyncReport> = None;
198    for round in 0..ROUNDS {
199        let first = round == 0;
200        let start = v.db().clone();
201        session.db = Some(start.clone());
202        let opts = SessionSyncOptions {
203            local_changed: Some(!first || options.local_changed),
204            accept_server_state: first && options.accept_server_state,
205            rekey_to: rekey.as_deref().map(|k| &k[..]).filter(|_| first),
206            remote_is_base: !first || remote_is_base,
207            cancel: Some(&cw),
208        };
209        let mut result = session.sync(&blob, &keys, opts).await;
210        if first && matches!(&result, Err(e) if e.is(OvErrorCode::VaultKeyMismatch)) {
211            // Another device rotated the vault key since this device fetched its keys: once more
212            // with the keys the server holds for this principal now
213            keys = refetch_keys(v, cancel).await?;
214            result = session.sync(&blob, &keys, opts).await;
215        }
216        v.inner.offline.store(session.offline, std::sync::atomic::Ordering::SeqCst);
217        if let VaultSource::Account { account, .. } = &v.inner.source {
218            account.note_clock(blob.clock_offset_ms());
219            if let Err(e) = &result {
220                if e.is(OvErrorCode::SignedOut) && e.status.is_some() {
221                    account.session_ended(e).await;
222                }
223                if e.is_unreachable() {
224                    account.set_offline(true);
225                }
226            } else {
227                account.set_offline(false);
228            }
229        }
230        let r = match result {
231            Ok(r) => r,
232            Err(e) if first => return Err(e),
233            Err(_) => {
234                // The edits made during the first rounds stay in the working copy, kept on the device
235                session.db = Some(v.db().clone());
236                let _ = session.keep_edits().await;
237                break;
238            }
239        };
240        *lock(&v.inner.etag) = Some(r.state.etag.clone());
241        total = Some(combine_reports(total, report(&r)));
242        let synced = session.db.clone().unwrap_or_else(|| r.db.clone());
243        let now = v.now();
244        let more = {
245            let mut db = v.db();
246            if *db == start {
247                *db = synced;
248                false
249            } else {
250                // Edited while the round ran: those edits on top of its result
251                let (mut merged, m) = ov_merge::merge(Some(&start), &db, &synced, now);
252                merged.header = synced.header.clone();
253                *db = merged.clone();
254                session.db = Some(merged);
255                m.remote_changed
256            }
257        };
258        if !more {
259            break;
260        }
261        if round + 1 == ROUNDS {
262            let _ = session.keep_edits().await;
263        }
264    }
265    let rep = total.expect("the first round succeeded");
266    *lock(&v.inner.conflicts) = rep.conflicts.clone();
267    Ok(rep)
268}

270fn outcome_rank(o: SyncOutcome) -> u8 {
271    match o {
272        SyncOutcome::Unchanged => 0,
273        SyncOutcome::Pulled => 1,
274        SyncOutcome::Uploaded => 2,
275        SyncOutcome::Created => 3,
276    }
277}

279/// The report of several rounds of one sync: the most a round did, every round's conflicts, changes
280/// and attempts, the last round's ETag and clock.
281fn combine_reports(so_far: Option<SyncReport>, next: SyncReport) -> SyncReport {
282    let Some(mut a) = so_far else { return next };
283    if outcome_rank(next.outcome) > outcome_rank(a.outcome) {
284        a.outcome = next.outcome;
285    }
286    for c in next.conflicts {
287        if !a.conflicts.contains(&c) {
288            a.conflicts.push(c);
289        }
290    }
291    a.changes.extend(next.changes);
292    a.attempts += next.attempts;
293    a.etag = next.etag.or(a.etag);
294    a.clock_offset_ms = next.clock_offset_ms.or(a.clock_offset_ms);
295    a.clock_skewed = next.clock_skewed;
296    a
297}

299/// Downloads (or, offline, opens this device's copy of) an account vault through a new session.
300pub(crate) async fn open_session(
301    account: &Account,
302    e: &super::keyring::VaultEntry,
303    cancel: &CancelToken,
304) -> Result<Vault, OvError> {
305    let v = super::vaults::new_vault(
306        VaultSource::Account { account: account.clone(), info: std::sync::Mutex::new(e.info.clone()) },
307        Some(account.host().clone()),
308        ov_kdbx::vault::create_kdbx("", 0)?,
309        false,
310        account.vault_cache(&e.info.id),
311    );
312    let client = account.sync_client();
313    let blob = client.vault(&e.info.id);
314    let cw = CancelWrap(cancel.clone());
315    let mut session = v.inner.session.lock().await;
316    let opts = SessionSyncOptions { cancel: Some(&cw), ..Default::default() };
317    let mut keys = e.keys().to_vec();
318    let mut first = session.sync(&blob, &keys, opts).await;
319    if matches!(&first, Err(err) if err.is(OvErrorCode::Forbidden)) {
320        // A vault shared since the token was issued isn't in its `vlt` claim yet: refresh once.
321        account.refresh_token(cancel).await?;
322        first = session.sync(&blob, &keys, opts).await;
323    }
324    if matches!(&first, Err(err) if err.is(OvErrorCode::VaultKeyMismatch)) {
325        // Another device rotated the vault key since this device fetched its keys: fetch them again
326        keys = refetch_keys(&v, cancel).await?;
327        first = session.sync(&blob, &keys, opts).await;
328    }
329    match first {
330        Ok(r) => {
331            account.note_clock(blob.clock_offset_ms());
332            account.set_offline(false);
333            *lock(&v.inner.etag) = Some(r.state.etag.clone());
334            *lock(&v.inner.conflicts) = report(&r).conflicts;
335        }
336        Err(err) => {
337            if err.is(OvErrorCode::SignedOut) && err.status.is_some() {
338                account.session_ended(&err).await;
339            }
340            if !(account.keeps_offline_data() && err.is_unreachable()) {
341                return Err(err);
342            }
343            session.open_local_copy(&keys, &e.info.name, &err).await?;
344            account.set_offline(true);
345        }
346    }
347    v.inner.offline.store(session.offline, std::sync::atomic::Ordering::SeqCst);
348    if let Some(db) = &session.db {
349        *v.db() = db.clone();
350    }
351    drop(session);
352    Ok(v)
353}

355/// Downloads a vault for a service account (no offline cache): one GET.
356pub(crate) async fn open_sa_session(
357    sa: &ServiceAccount,
358    e: &super::keyring::VaultEntry,
359    cancel: &CancelToken,
360) -> Result<Vault, OvError> {
361    let v = super::vaults::new_vault(
362        VaultSource::ServiceAccount { sa: Arc::downgrade(&sa.inner), info: e.info.clone() },
363        Some(sa.inner.client.host.clone()),
364        ov_kdbx::vault::create_kdbx("", 0)?,
365        false,
366        None,
367    );
368    let client = sa.sync_client();
369    let blob = client.vault(&e.info.id);
370    let cw = CancelWrap(cancel.clone());
371    let mut session = v.inner.session.lock().await;
372    // One GET: the downloaded copy is the working copy and the base (TS `ServiceAccount.download`)
373    let r = match session.download(&blob, e.keys(), Some(&cw)).await {
374        Err(err) if err.is(OvErrorCode::VaultKeyMismatch) => {
375            // The vault key was rotated since the list was fetched: fetch it again, once
376            let keys = refetch_keys(&v, cancel).await?;
377            session.download(&blob, &keys, Some(&cw)).await?
378        }
379        r => r?,
380    };
381    *lock(&v.inner.etag) = Some(r.state.etag.clone());
382    if let Some(db) = &session.db {
383        *v.db() = db.clone();
384    }
385    drop(session);
386    Ok(v)
387}

389/// One sync of a vault from its pending edits or the server's copy, re-keyed to `new_key` before
390/// the upload (rotation). `e` must already hold the new key.
391pub(crate) async fn rekey_sync(
392    account: &Account,
393    e: &super::keyring::VaultEntry,
394    new_key: &[u8],
395    cancel: &CancelToken,
396) -> Result<SyncReport, OvError> {
397    let client = account.sync_client();
398    let blob = client.vault(&e.info.id);
399    let cw = CancelWrap(cancel.clone());
400    let mut session = ov_sync::VaultSession::new(account.vault_cache(&e.info.id));
401    let opts = SessionSyncOptions {
402        local_changed: Some(true),
403        accept_server_state: false,
404        rekey_to: Some(new_key),
405        remote_is_base: false,
406        cancel: Some(&cw),
407    };
408    let r = session.sync(&blob, e.keys(), opts).await;
409    account.note_clock(blob.clock_offset_ms());
410    Ok(report(&r?))
411}

413/// Keeps the working copy as pending edits (Swift `keepEdits`).
414pub(crate) async fn keep_edits(v: &Vault) -> Result<(), OvError> {
415    if !matches!(v.inner.source, VaultSource::Account { .. }) {
416        return Ok(());
417    }
418    let (_, keys) = vault_keys(v).await?;
419    let mut session = v.inner.session.lock().await;
420    session.db = Some(v.db().clone());
421    if session.key.is_none() {
422        if let Some(k) = keys.iter().max_by_key(|k| k.version) {
423            session.key = Some(vault_credentials(&k.key)?);
424        }
425    }
426    session.keep_edits().await
427}

429/// Self-hosted whole-database sync (SPEC §13), the state kept in the BlobCache under the user
430/// `selfhosted:<url>`. With `local_changed: false` and a stored state, a HEAD first: unchanged on the
431/// server means nothing is decrypted, merged or downloaded, and `kdbx` comes back as it is.
432pub(crate) async fn sync_database(
433    sh: &SelfHosted,
434    database_id: &str,
435    kdbx: &[u8],
436    credentials: &KdbxCredentials,
437    options: SyncOptions,
438    cancel: &CancelToken,
439) -> Result<SelfHostedSyncResult, OvError> {
440    cancel.check()?;
441    let host = &sh.inner.client.host;
442    let key = super::vaults::local_key(credentials)?;
443    let client = SyncClient::new(
444        host.http.clone(),
445        host.clock.clone(),
446        sh.inner.sync.clone(),
447        Arc::new(ov_sync::StaticToken(Some(sh.inner.token.to_string()))),
448    );
449    let blob = client.database(database_id);
450    let cache = VaultCache::new(host.cache.clone(), super::self_hosted::token_key(&sh.inner.server_url), database_id);
451    let state = if options.accept_server_state { None } else { cache.load_state().await? };
452    let mut head_changed = false;
453    if let (Some(prev), false) = (&state, options.local_changed) {
454        // Before the file is decrypted (a KDF run): most checks find nothing new
455        if let ov_protocol::sync::HeadResult::NotModified(v) = blob.head(Some(&prev.etag)).await? {
456            if let Some(r) = v.revision.filter(|r| *r >= 0 && *r < prev.revision) {
457                return Err(OvError::new(
458                    OvErrorCode::ServerRolledBack,
459                    format!(
460                        "the sync server returned revision {r}, older than revision {} this device already synced",
461                        prev.revision
462                    ),
463                ));
464            }
465            let mut checked = prev.clone();
466            checked.last_sync = ov_sync::state::iso_time(host.clock.now_unix_ms());
467            cache.save_state(&checked).await?;
468            let offset = blob.clock_offset_ms();
469            let report = SyncReport {
470                outcome: SyncOutcome::Unchanged,
471                conflicts: Vec::new(),
472                changes: Vec::new(),
473                attempts: 0,
474                etag: Some(prev.etag.clone()).filter(|e| !e.is_empty()),
475                clock_offset_ms: offset,
476                clock_skewed: ov_sync::is_clock_skewed(offset),
477            };
478            return Ok(SelfHostedSyncResult { kdbx: kdbx.to_vec(), report });
479        }
480        head_changed = true;
481    }
482    cancel.check()?;
483    let mut db = ov_kdbx::read(kdbx, &key)?;
484    let creds = [key.clone()];
485    let cw = CancelWrap(cancel.clone());
486    // After a HEAD that found a new version, don't ask again: download and merge
487    let mut opts = ov_sync::SyncOptions::new(&key, &creds)
488        .state(state.as_ref())
489        .local_changed(options.local_changed || head_changed)
490        .accept_server_state(options.accept_server_state);
491    opts.cancel = Some(&cw);
492    let r = ov_sync::sync_kdbx(&blob, &mut db, &opts).await?;
493    cache.save_state(&r.state).await?;
494    let out = ov_kdbx::write(&r.db, &r.key)?;
495    Ok(SelfHostedSyncResult { kdbx: out, report: report(&r) })
496}

512/// The next event, after the account acted on it (`access`: refresh, reload the vault list,
513/// reconnect; deletion: end the session and close).
514pub(crate) async fn next_event(s: &EventStream, cancel: &CancelToken) -> Result<Option<AccountEvent>, OvError> {
515    loop {
516        cancel.check()?;
517        let item = {
518            let fut = s.inner.stream.next();
519            let c = cancel.cancelled();
520            futures::pin_mut!(fut, c);
521            match futures::future::select(fut, c).await {
522                futures::future::Either::Left((i, _)) => i,
523                futures::future::Either::Right(_) => return Err(OvError::new(OvErrorCode::Aborted, "cancelled")),
524            }
525        };
526        match item {
527            StreamItem::Connecting | StreamItem::Open => continue,
528            StreamItem::Closed { .. } => return Ok(None),
529            StreamItem::Waiting { retry_in_ms, error } => {
530                let base = error.unwrap_or_else(|| OvError::new(OvErrorCode::NetworkError, "the event stream ended"));
531                let mut e = OvError::new(OvErrorCode::NetworkError, base.message.clone());
532                e.status = base.status;
533                let secs = retry_in_ms.div_ceil(1000).min(u64::from(u32::MAX)) as u32;
534                return Err(e.with_retry_after(secs).with_details(format!("{{\"retryInMs\":{retry_in_ms}}}")));
535            }
536            StreamItem::Event(ev) => match ev {
537                SyncEvent::Vault { id, etag, revision, deleted } => {
538                    return Ok(Some(AccountEvent::VaultChanged {
539                        vault_id: id,
540                        etag,
541                        revision: rev(revision),
542                        deleted,
543                    }))
544                }
545                SyncEvent::Database { id, etag, revision, deleted } => {
546                    return Ok(Some(AccountEvent::DatabaseChanged {
547                        database_id: id,
548                        etag,
549                        revision: rev(revision),
550                        deleted,
551                    }))
552                }
553                SyncEvent::Access { change, at, .. } => {
554                    let Some(account) = &s.inner.account else { continue };
555                    let reason = match change.as_str() {
556                        "deletion_scheduled" => Some(SignedOutReason::DeletionScheduled),
557                        "account_deleted" => Some(SignedOutReason::AccountDeleted),
558                        _ => None,
559                    };
560                    if let Some(reason) = reason {
561                        s.inner.stream.close();
562                        if let Some(user) = account.user_id() {
563                            account.end_session(&user, reason).await?;
564                        }
565                        return Ok(Some(AccountEvent::SessionEnded { reason, at }));
566                    }
567                    account.refresh_token(cancel).await?;
568                    let vaults = account.load_vault_list(cancel).await?.0;
569                    s.inner.stream.reconnect();
570                    return Ok(Some(AccountEvent::VaultsChanged { vaults }));
571                }
572                SyncEvent::Unknown { .. } => continue,
573            },
574        }
575    }
576}
