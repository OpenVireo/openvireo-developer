1//! Account vaults held as `.kdbx` bytes by the platform (the desktop keeps its KeePassXC `Database`
2//! and serializes it), contract-v1.6: the vault key as file credentials, and a whole-file sync on
3//! `ov_sync::sync_kdbx`, as [`sync_database`](super::syncing::sync_database) does for self-hosted
4//! databases. The sync state lives in the BlobCache record `kdbx-file:<vault id>` when the account
5//! keeps offline data.
6
7use ov_error::{OvError, OvErrorCode};
8use ov_kdbx::vault::{open_with_key_versions, vault_credentials, KeyVersion};

16/// Whose vault file: an account's (offline cache, session end) or a service account's (contract-v1.8).
17enum Owner<'a> {
18    Account(&'a Account),
19    Sa(&'a ServiceAccount),
20}
21
22impl Owner<'_> {
23    async fn entry(&self, vault_id: &str, cancel: &CancelToken) -> Result<VaultEntry, OvError> {
24        match self {
25            Owner::Account(a) => a.entry(vault_id, cancel).await,
26            Owner::Sa(sa) => sa.entry(vault_id, cancel).await,
27        }
28    }
29
30    fn client(&self) -> ov_sync::SyncClient {
31        match self {
32            Owner::Account(a) => a.sync_client(),
33            Owner::Sa(sa) => sa.sync_client(),
34        }
35    }
36
37    async fn refresh(&self, cancel: &CancelToken) -> Result<(), OvError> {
38        match self {
39            Owner::Account(a) => a.refresh_token(cancel).await,
40            Owner::Sa(sa) => sa.access_token(true, cancel).await.map(|_| ()),
41        }
42    }
43}
44
45fn file_credentials(k: &KeyVersion) -> KdbxCredentials {
46    KdbxCredentials { password: None, key_file: Some(k.key.to_vec()), password_hash: None }
47}

49pub(crate) async fn credentials(a: &Account, vault_id: &str, cancel: &CancelToken) -> Result<KdbxCredentials, OvError> {
50    owner_credentials(Owner::Account(a), vault_id, cancel).await
51}

56    cancel: &CancelToken,

58    owner_credentials(Owner::Sa(sa), vault_id, cancel).await

61async fn owner_credentials(o: Owner<'_>, vault_id: &str, cancel: &CancelToken) -> Result<KdbxCredentials, OvError> {
62    cancel.check()?;
63    let e = o.entry(vault_id, cancel).await?;
64    Ok(file_credentials(newest_key(e.keys())?))
65}

89async fn owner_sync(
90    o: Owner<'_>,
91    vault_id: &str,
92    kdbx: Vec<u8>,
93    base: Option<Vec<u8>>,
94    options: SyncOptions,
95    cancel: &CancelToken,
96) -> Result<VaultKdbxSyncResult, OvError> {
97    cancel.check()?;
98    let e = o.entry(vault_id, cancel).await?;
99    let keys = e.keys().to_vec();
100    let (db, _) = open_with_key_versions(&kdbx, &keys)?;
101    drop(kdbx);
102    let mut newest: Vec<&KeyVersion> = keys.iter().collect();
103    newest.sort_by_key(|k| std::cmp::Reverse(k.version));
104    let creds = newest.iter().map(|k| vault_credentials(&k.key)).collect::<Result<Vec<_>, _>>()?;
105    let local_key = creds.first().ok_or_else(|| OvError::new(OvErrorCode::NoVaultKey, "no vault key available"))?;
106
107    // Its own record: the file and a `Vault` handle of the same vault are separate working copies.
108    let cache = match &o {
109        Owner::Account(a) => a.vault_cache(&format!("kdbx-file:{vault_id}")),
110        Owner::Sa(_) => None,
111    };
112    let mut state = match &cache {
113        Some(c) => c.load_state().await?,
114        None => None,
115    };
116    if let Some(b) = base {
117        match &mut state {
118            Some(s) => s.base = Some(b),
119            None => {
120                state = Some(SyncState {
121                    etag: String::new(),
122                    revision: -1,
123                    generation: 0,
124                    snapshot: Default::default(),
125                    base: Some(b),
126                    last_sync: String::new(),
127                })
128            }
129        }
130    }
131
132    let mut attempt = 0;
133    let r = loop {
134        let client = o.client();
135        let blob = client.vault(vault_id);
136        let cw = CancelWrap(cancel.clone());
137        let mut opts = ov_sync::SyncOptions::new(local_key, &creds)
138            .state(state.as_ref())
139            .local_changed(options.local_changed)
140            .accept_server_state(options.accept_server_state);
141        opts.cancel = Some(&cw);
142        let mut work = db.clone();
143        let result = ov_sync::sync_kdbx(&blob, &mut work, &opts).await;
144        if let Owner::Account(a) = &o {
145            a.note_clock(blob.clock_offset_ms());
146        }
147        match result {
148            Err(err) if err.is(OvErrorCode::Forbidden) && attempt == 0 => {
149                // A vault shared since the token was issued isn't in its `vlt` claim yet: refresh once.
150                attempt += 1;
151                o.refresh(cancel).await?;
152            }
153            Err(err) => {
154                if let Owner::Account(a) = &o {
155                    if err.is(OvErrorCode::SignedOut) && err.status.is_some() {
156                        a.session_ended(&err).await;
157                    }
158                    if err.is_unreachable() {
159                        a.set_offline(true);
160                    }
161                }
162                return Err(err);
163            }
164            Ok(r) => {
165                if let Owner::Account(a) = &o {
166                    a.set_offline(false);
167                }
168                break r;
169            }
170        }
171    };
172    if let Some(c) = &cache {
173        c.save_state(&r.state).await?;
174    }
175    let used = r.credentials_index.and_then(|i| newest.get(i).copied()).unwrap_or(newest[0]);
176    let out = ov_kdbx::write(&r.db, &r.key)?;
177    Ok(VaultKdbxSyncResult { kdbx: out, credentials: file_credentials(used), report: report(&r) })
178}
