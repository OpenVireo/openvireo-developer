1//! Helpers behind [`Vault`]: the clock, the source checks and the working copy lock.
2
3use std::sync::MutexGuard;
4
5use ov_error::{OvError, OvErrorCode};
6use ov_kdbx::model::{Database, Timestamp};
7
8use super::util::lock;
9use crate::api::{Vault, VaultInfo};
10use crate::state::VaultSource;
11
12impl Vault {
13    pub(crate) fn db(&self) -> MutexGuard<'_, Database> {
14        lock(&self.inner.db)
15    }
16
17    /// KDBX time (Unix seconds) from the Host clock; for a local file opened without a Host, the
18    /// system clock (native) or one second after the newest time in the file (wasm32, which has no
19    /// clock of its own).
20    pub(crate) fn now(&self) -> Timestamp {
21        if let Some(h) = &self.inner.host {
22            return h.clock.now_unix_ms().div_euclid(1000);
23        }
24        #[cfg(not(target_arch = "wasm32"))]
25        {
26            std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).map_or(0, |d| d.as_secs() as i64)
27        }
28        #[cfg(target_arch = "wasm32")]
29        {
30            let db = self.db();
31            db.root.times.last_modification.max(db.meta.settings_changed.unwrap_or(0)) + 1
32        }
33    }
34
35    /// `invalid_request` for an old version from history.
36    pub(crate) fn writable(&self) -> Result<(), OvError> {
37        if matches!(self.inner.source, VaultSource::Version { .. }) {
38            return Err(OvError::new(OvErrorCode::InvalidRequest, "an old version is read-only"));
39        }
40        Ok(())
41    }
42
43    /// Edits the working copy (refused for read-only versions).
44    pub(crate) fn edit<T>(&self, f: impl FnOnce(&mut Database, Timestamp) -> Result<T, OvError>) -> Result<T, OvError> {
45        self.writable()?;
46        let now = self.now();
47        f(&mut self.db(), now)
48    }
49
50    pub(crate) fn vault_info(&self) -> Option<VaultInfo> {
51        match &self.inner.source {
52            VaultSource::Local(_) => None,
53            VaultSource::Account { info, .. } => Some(lock(info).clone()),
54            VaultSource::Version { info } => info.clone(),
55            VaultSource::ServiceAccount { info, .. } => Some(info.clone()),
56        }
57    }
58}

62    /// Local files and versions keep nothing.
63    pub(crate) async fn keep_edits_impl(&self) -> Result<(), OvError> {
64        super::syncing::keep_edits(self).await
