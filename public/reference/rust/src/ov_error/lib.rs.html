1//! `OvError`, the one error type every openvireo-core API returns.
2//!
3//! [`OvErrorCode`] lists the TypeScript SDK's `OpenVireoErrorCode` values one for one
4//! (`packages/sdk/src/errors.ts`), plus the codes the SDK throws without listing them and
5//! [`OvErrorCode::Internal`] for a panic caught at a binding edge. Every code has a stable wire
6//! string (`"plan_limit"`) that bindings expose, so UI code written against the TS SDK keeps
7//! matching on the same strings.
8//!
9//! Server errors use the `{error, code}` body of `openapi.yaml`. A server code that isn't one of
10//! these variants keeps its exact string in [`OvError::server_code`], and the variant comes from
11//! the HTTP status (the SDK's `codeForStatus`), so [`OvError::wire_code`] still returns what the
12//! server sent.
13

20macro_rules! codes {
21    ($( $(#[$doc:meta])* $variant:ident => $wire:literal, )+) => {
22        /// A machine-readable error code. [`OvErrorCode::as_str`] is the stable wire string.
23        #[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
24        #[non_exhaustive]
25        pub enum OvErrorCode {
26            $( $(#[$doc])* $variant, )+
27        }
28
29        impl OvErrorCode {
30            /// Every code, in declaration order.
31            pub const ALL: &'static [OvErrorCode] = &[ $( OvErrorCode::$variant, )+ ];
32
33            /// The stable wire string, e.g. `"plan_limit"`.
34            pub const fn as_str(self) -> &'static str {
35                match self { $( OvErrorCode::$variant => $wire, )+ }
36            }
37
38            /// The code for a wire string, or `None` if it isn't one of these codes.
39            pub fn from_wire(s: &str) -> Option<Self> {
40                match s { $( $wire => Some(OvErrorCode::$variant), )+ _ => None }
41            }
42        }
43    };
44}

168impl OvErrorCode {
169    /// The code the TS SDK uses for an HTTP status when the body names no known code
170    /// (`codeForStatus` in `http/request.ts`).
171    pub const fn for_status(status: u16) -> Self {
172        match status {
173            400 => OvErrorCode::InvalidRequest,
174            401 => OvErrorCode::Unauthorized,
175            402 => OvErrorCode::PlanLimit,
176            403 => OvErrorCode::Forbidden,
177            404 => OvErrorCode::NotFound,
178            409 => OvErrorCode::Conflict,
179            410 => OvErrorCode::Gone,
180            412 => OvErrorCode::PreconditionFailed,
181            428 => OvErrorCode::PreconditionRequired,
182            429 => OvErrorCode::RateLimited,
183            s if s >= 500 => OvErrorCode::ServerError,
184            _ => OvErrorCode::InvalidRequest,
185        }
186    }
187}

195/// The wire string isn't one of the [`OvErrorCode`] values.
196#[derive(Clone, Debug, PartialEq, Eq)]
197pub struct UnknownCode(pub String);

199impl fmt::Display for UnknownCode {
200    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
201        write!(f, "unknown error code `{}`", self.0)
202    }
203}
204
205impl std::error::Error for UnknownCode {}

207impl FromStr for OvErrorCode {
208    type Err = UnknownCode;
209    fn from_str(s: &str) -> Result<Self, Self::Err> {
210        OvErrorCode::from_wire(s).ok_or_else(|| UnknownCode(s.to_string()))
211    }
212}

214/// An error from openvireo-core. Every fallible API returns `Result<T, OvError>`.
215///
216/// `message` is for people and may change; match on [`OvError::code`] or [`OvError::wire_code`].
217/// Messages never contain secrets.
218#[derive(Clone, Debug, PartialEq, Eq)]
219pub struct OvError {
220    /// What went wrong.
221    pub code: OvErrorCode,
222    /// A human-readable description (English, not localised).
223    pub message: String,
224    /// The HTTP status, when the error came from a server response.
225    pub status: Option<u16>,
226    /// Seconds to wait, from a `Retry-After` header (429, 503).
227    pub retry_after: Option<u32>,
228    /// Extra data some codes carry, as a JSON object with camelCase keys, matching the TS SDK's
229    /// `OpenVireoError.details`: `recovery_not_ready` → `notBefore`; `plan_limit` → `limit`, `max`,
230    /// `current`, `plan`, `upgradeUrl`; `quota_exceeded` → `max`; `invalid_passcode` →
231    /// `attemptsLeft`; a 409 `deletion_blocked` → `orgs`.
232    pub details: Option<String>,
233    /// The server's `code` when it isn't one of the [`OvErrorCode`] wire strings (for example
234    /// `deletion_blocked` or `confirm_mismatch`). `code` then comes from the HTTP status.
235    pub server_code: Option<String>,
236}
237
238impl OvError {
239    /// An error with a code and message and nothing else.
240    pub fn new(code: OvErrorCode, message: impl Into<String>) -> Self {
241        OvError { code, message: message.into(), status: None, retry_after: None, details: None, server_code: None }
242    }
243
244    /// An [`OvErrorCode::Internal`] error.
245    pub fn internal(message: impl Into<String>) -> Self {
246        OvError::new(OvErrorCode::Internal, message)
247    }
248
249    /// An error built from a server's `{error, code}` body: a known `code` is used as is; an
250    /// unknown one is kept in `server_code` and the variant comes from the status.
251    pub fn from_server(status: u16, code: Option<&str>, message: impl Into<String>) -> Self {
252        let (variant, server_code) = match code {
253            Some(c) => match OvErrorCode::from_wire(c) {
254                Some(v) => (v, None),
255                None => (OvErrorCode::for_status(status), Some(c.to_string())),
256            },
257            None => (OvErrorCode::for_status(status), None),
258        };
259        OvError { server_code, ..OvError::new(variant, message).with_status(status) }
260    }
261
262    /// Sets the HTTP status.
263    pub fn with_status(mut self, status: u16) -> Self {
264        self.status = Some(status);
265        self
266    }
267
268    /// Sets `Retry-After` seconds.
269    pub fn with_retry_after(mut self, seconds: u32) -> Self {
270        self.retry_after = Some(seconds);
271        self
272    }
273
274    /// Sets the details JSON object.
275    pub fn with_details(mut self, details_json: impl Into<String>) -> Self {
276        self.details = Some(details_json.into());
277        self
278    }
279
280    /// The code as a string, as the TS SDK's `err.code` would hold it: the server's own code when
281    /// it isn't a known variant, else [`OvErrorCode::as_str`].
282    pub fn wire_code(&self) -> &str {
283        self.server_code.as_deref().unwrap_or(self.code.as_str())
284    }
285
286    /// Whether this error has `code`, like the SDK's `isOpenVireoError(e, code)`.
287    pub fn is(&self, code: OvErrorCode) -> bool {
288        self.code == code
289    }
290
291    /// Whether the server couldn't be reached (no network, or a gateway answering 502/503/504), as
292    /// opposed to refusing the request. The SDK's `isUnreachable`; offline fallbacks (SPEC §20) use it.
293    pub fn is_unreachable(&self) -> bool {
294        self.code == OvErrorCode::NetworkError || matches!(self.status, Some(502..=504))
295    }
296}
297
298impl fmt::Display for OvError {
299    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
300        write!(f, "{}: {}", self.wire_code(), self.message)
301    }
302}

306impl From<OvErrorCode> for OvError {

308        OvError::new(code, code.as_str())
309    }

315#[cfg(test)]
316mod tests {
317    use super::*;
318
319    /// The §2.4 list from rust-core-tasks.md, in order: the TS SDK's codes.
320    const TS_CODES: &str =
321        "unsupported_version decryption_failed invalid_secret_key invalid_public_keys invalid_token \
322        weak_kdf low_order_point opaque_failed invalid_request unauthorized forbidden not_found conflict rate_limited \
323        mfa_required invalid_mfa mfa_expired session_expired token_reused precondition_failed precondition_required \
324        gone server_error network_error insecure_url signed_out recovery_not_ready recovery_cancelled restore_locked \
325        invalid_proof plan_limit quota_exceeded billed_by_stripe apple_subscription_bound apple_verification_failed \
326        vault_key_mismatch no_vault_key kdbx_error merge_failed server_rolled_back server_lost_database \
327        too_many_conflicts invalid_reference entry_not_found not_available_offline vault_upload_failed aborted";
328
329    #[test]
330    fn every_ts_code_has_a_variant_in_order() {
331        let ts: Vec<&str> = TS_CODES.split_whitespace().collect();
332        let ours: Vec<&str> = OvErrorCode::ALL.iter().map(|c| c.as_str()).collect();
333        assert_eq!(&ours[..ts.len()], &ts[..]);
334        assert_eq!(ours.last(), Some(&"internal"));
335    }
336
337    #[test]
338    fn wire_strings_round_trip_and_are_unique() {
339        let mut seen = std::collections::HashSet::new();
340        for &c in OvErrorCode::ALL {
341            assert!(seen.insert(c.as_str()), "duplicate {c}");
342            assert_eq!(c.as_str().parse::<OvErrorCode>(), Ok(c));
343        }
344        assert!("nope".parse::<OvErrorCode>().is_err());
345    }
346
347    #[test]
348    fn server_errors_keep_unknown_codes() {
349        let e = OvError::from_server(409, Some("deletion_blocked"), "hand over the orgs first");
350        assert_eq!(e.code, OvErrorCode::Conflict);
351        assert_eq!(e.wire_code(), "deletion_blocked");
352        let e = OvError::from_server(402, Some("plan_limit"), "limit");
353        assert_eq!((e.code, e.server_code, e.status), (OvErrorCode::PlanLimit, None, Some(402)));
354        assert_eq!(OvError::from_server(503, None, "x").code, OvErrorCode::ServerError);
355        assert!(OvError::from_server(503, None, "x").is_unreachable());
356        assert_eq!(OvErrorCode::for_status(418), OvErrorCode::InvalidRequest);
357    }
