1//! The Host: everything the core asks the platform to do (rust-core.md §5).
2//!
3//! **Frozen contract** (rust-core-tasks.md §2.2, tag `contract-v1`; moved unchanged from
4//! `ov_core::host` into this crate in `contract-v1.5` so ov-sync and ov-vault can use it without
5//! depending on ov-core, which re-exports it as `ov_core::host`). The core never opens a socket
6//! or a file, reads the clock or touches the Keychain itself. Each platform implements these traits
7//! once (URLSession + Keychain on iOS, QNetworkAccessManager + DPAPI/libsecret on desktop, `fetch` +
8//! IndexedDB on the web, reqwest + files in the `ov` CLI) and passes a [`Host`] to
9//! `ov_core::api::OvClient::new`.
10//!
11//! Randomness is not part of the Host: the core reads the platform CSPRNG through `getrandom`
12//! (`crypto.getRandomValues` on wasm32).
13//!
14//! Async traits use `async_trait`. On wasm32 the futures and trait objects are not `Send`
15//! (`async_trait(?Send)`, and [`MaybeSend`]/[`MaybeSync`] are empty), because JS values can't
16//! cross threads; everywhere else they are `Send + Sync`.
17

54/// Why a request didn't produce a response.
55#[derive(Clone, Debug, PartialEq, Eq)]
56pub enum TransportError {
57    /// No network, DNS failure or connection refused.
58    Offline,
59    /// The request timed out.
60    Timeout,
61    /// TLS failed (certificate, handshake); the text is for logs.
62    Tls(String),
63    /// The platform cancelled the request.
64    Cancelled,
65    /// Anything else; the text is for logs.
66    Other(String),
67}
68
69impl From<TransportError> for OvError {
70    /// `Cancelled` becomes `aborted`; everything else is `network_error`, which offline fallbacks
71    /// (SPEC §20) treat as "the server can't be reached".
72    fn from(e: TransportError) -> Self {
73        match e {
74            TransportError::Cancelled => OvError::new(OvErrorCode::Aborted, "the request was cancelled"),
75            TransportError::Offline => OvError::new(OvErrorCode::NetworkError, "the server can't be reached"),
76            TransportError::Timeout => OvError::new(OvErrorCode::NetworkError, "the request timed out"),
77            TransportError::Tls(m) => OvError::new(OvErrorCode::NetworkError, format!("TLS failed: {m}")),
78            TransportError::Other(m) => OvError::new(OvErrorCode::NetworkError, m),
79        }
80    }
81}

83/// HTTP for the core: URLSession, QNetworkAccessManager, `fetch` or reqwest.
84#[cfg_attr(not(target_arch = "wasm32"), async_trait::async_trait)]
85#[cfg_attr(target_arch = "wasm32", async_trait::async_trait(?Send))]
86pub trait HttpTransport: MaybeSend + MaybeSync {
87    /// Sends one request and returns the whole response.
88    async fn send(&self, req: HttpRequest) -> Result<HttpResponse, TransportError>;
89
90    /// Opens a server-sent events stream (SPEC §11): `GET {sync}/events` with `Accept:
91    /// text/event-stream`. The transport checks the status (an error status may be returned as
92    /// `Ok` with a stream that yields the body); the core parses the SSE framing. Browsers must
93    /// use `fetch` with `cache: "no-store"`, not `EventSource`.
94    async fn open_stream(&self, req: HttpRequest) -> Result<Box<dyn ByteStream>, TransportError>;
95}
96
97/// The body of an open stream, read chunk by chunk.
98#[cfg_attr(not(target_arch = "wasm32"), async_trait::async_trait)]
99#[cfg_attr(target_arch = "wasm32", async_trait::async_trait(?Send))]
100pub trait ByteStream: MaybeSend + MaybeSync {
101    /// The response status the stream was opened with.
102    fn status(&self) -> u16;
103    /// The next chunk of bytes (any size), or `None` when the server closed the stream.
104    async fn next_chunk(&self) -> Result<Option<Vec<u8>>, TransportError>;
105    /// Closes the stream; later `next_chunk` calls return `None`.
106    fn close(&self);
107}

111/// Small secrets that must survive restarts: sessions (refresh tokens), trusted Secret Keys,
112/// device ids, the remembered-accounts list (SPEC §2, §21). Keychain, DPAPI/libsecret, or
113/// IndexedDB values wrapped with a non-extractable WebCrypto key.
114///
115/// Keys are short ASCII strings chosen by the core (`secret_key:<idS>:<idU>`, `device:…`,
116/// `session`, `parked_session:…`; the same names the TS SDK and iOS app use, so existing data
117/// carries over). Values are opaque bytes. Implementations must be safe to call concurrently.
118#[cfg_attr(not(target_arch = "wasm32"), async_trait::async_trait)]
119#[cfg_attr(target_arch = "wasm32", async_trait::async_trait(?Send))]
120pub trait SecureStore: MaybeSend + MaybeSync {
121    /// The value for `key`, or `None`. A store that can't be read (e.g. a locked Keychain before
122    /// first unlock) returns an error, which is not the same as `None`.
123    async fn get(&self, key: &str) -> Result<Option<Zeroizing<Vec<u8>>>, OvError>;
124    /// Stores `value` under `key`, replacing any old value.
125    async fn set(&self, key: &str, value: &[u8]) -> Result<(), OvError>;
126    /// Removes `key`; removing a missing key is not an error.
127    async fn delete(&self, key: &str) -> Result<(), OvError>;
128}

130/// What the offline cache keeps for one vault of one user (SPEC §20). Everything in it is already
131/// encrypted (vault bytes under the vault key) or not secret, so it can live in ordinary app
132/// storage (files, IndexedDB).
133#[derive(Clone, PartialEq, Eq)]
134pub struct CachedVault {
135    /// The core's sync state for the vault, serialized by the core (opaque to the host). It holds
136    /// the ETag, revision, sync generation, entry snapshot and last-sync time.
137    pub state: Vec<u8>,
138    /// The server's encrypted `.kdbx` at the last sync: the merge base and the offline copy.
139    pub base: Option<Vec<u8>>,
140    /// Local edits not yet uploaded: a `.kdbx` encrypted with the vault key.
141    pub pending: Option<Vec<u8>>,
142    /// Edits made by an AutoFill extension, waiting for the app to merge them (iOS), encrypted
143    /// like `pending`.
144    pub autofill: Option<Vec<u8>>,
145}
146
147impl std::fmt::Debug for CachedVault {
148    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
149        f.debug_struct("CachedVault")
150            .field("state_len", &self.state.len())
151            .field("base_len", &self.base.as_ref().map(Vec::len))
152            .field("pending_len", &self.pending.as_ref().map(Vec::len))
153            .field("autofill_len", &self.autofill.as_ref().map(Vec::len))
154            .finish()
155    }
156}

158/// The offline cache (SPEC §20): per user, per vault, plus a few per-user and per-server records.
159///
160/// Records are keyed by `user_id` because two accounts on one device can share a vault (SPEC §21).
161/// The `meta_*` records hold what offline unlock and listing need: the discovery document (`user_id`
162/// is then `""` and `name` is `config:<issuer>`), and per user the wrapped keyset + KDF (`profile`)
163/// and the vault list as the server returned it (`vaults`). All of it is ciphertext or public.
164#[cfg_attr(not(target_arch = "wasm32"), async_trait::async_trait)]
165#[cfg_attr(target_arch = "wasm32", async_trait::async_trait(?Send))]
166pub trait BlobCache: MaybeSend + MaybeSync {
167    /// The cached vault, or `None`.
168    async fn load(&self, user_id: &str, vault_id: &str) -> Result<Option<CachedVault>, OvError>;
169    /// Stores the cached vault, replacing the old record.
170    async fn save(&self, user_id: &str, vault_id: &str, v: &CachedVault) -> Result<(), OvError>;
171    /// Removes one vault's record.
172    async fn remove(&self, user_id: &str, vault_id: &str) -> Result<(), OvError>;
173    /// The ids of every vault cached for `user_id`.
174    async fn list(&self, user_id: &str) -> Result<Vec<String>, OvError>;
175    /// A per-user (or, with `user_id == ""`, per-device) record such as `profile` or `vaults`.
176    async fn load_meta(&self, user_id: &str, name: &str) -> Result<Option<Vec<u8>>, OvError>;
177    /// Stores a per-user record.
178    async fn save_meta(&self, user_id: &str, name: &str, value: &[u8]) -> Result<(), OvError>;
179    /// Removes everything cached for `user_id`: its vaults and its records (sign-out, SPEC §20).
180    async fn clear_user(&self, user_id: &str) -> Result<(), OvError>;
181}

191/// Log severity.
192#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
193pub enum Level {
194    /// Something failed.
195    Error,
196    /// Something unexpected that the core recovered from.
197    Warn,
198    /// Notable steps (sync outcomes, offline fallbacks).
199    Info,
200    /// Detail for debugging.
201    Debug,
202    /// Very detailed tracing.
203    Trace,
204}

206/// Where the core's log lines go (os_log, qDebug, `console`, stderr). The core never logs secrets.
207pub trait LogSink: MaybeSend + MaybeSync {
208    /// One log line. `target` names the module, e.g. `"ov_sync"`.
209    fn log(&self, level: Level, target: &str, msg: &str);
210}

212/// The platform services the core runs on. Cheap to clone; every `ov_core::api::OvClient`
213/// and handle made from it shares these.
214#[derive(Clone)]
215pub struct Host {
216    /// HTTP.
217    pub http: Arc<dyn HttpTransport>,
218    /// Secrets that persist.
219    pub store: Arc<dyn SecureStore>,
220    /// The offline cache.
221    pub cache: Arc<dyn BlobCache>,
222    /// The clock.
223    pub clock: Arc<dyn Clock>,
224    /// Logs.
225    pub log: Arc<dyn LogSink>,
226}
227
228impl std::fmt::Debug for Host {
229    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
230        f.write_str("Host { .. }")
231    }
